5/// Collects all commands into a [`Vec<serenity::CreateCommand>`] builder, which can be used
6/// to register the commands on Discord
7///
8/// Also see [`register_application_commands_buttons`] for a ready to use register command
9///
10/// ```rust,no_run
11/// # use poise::serenity_prelude as serenity;
12/// # async fn foo(ctx: poise::Context<'_, (), ()>) -> Result<(), serenity::Error> {
13/// let commands = &ctx.framework().options().commands;
14/// let create_commands = poise::builtins::create_application_commands(commands);
15///
16/// serenity::Command::set_global_commands(ctx, create_commands).await?;
17/// # Ok(()) }
18/// ```
19pub fn create_application_commands<'a, U: 'a, E: 'a>(
20    commands: impl IntoIterator<Item = &'a crate::Command<U, E>>,
21) -> Vec<serenity::CreateCommand> {
22    /// We decided to extract context menu commands recursively, despite the subcommand hierarchy
23    /// not being preserved. Because it's more confusing to just silently discard context menu
24    /// commands if they're not top-level commands.
25    /// https://discord.com/channels/381880193251409931/919310428344029265/947970605985189989
26    fn recursively_add_context_menu_commands<U, E>(
27        builder: &mut Vec<serenity::CreateCommand>,
28        command: &crate::Command<U, E>,
29    ) {
30        if let Some(context_menu_command) = command.create_as_context_menu_command() {
31            builder.push(context_menu_command);
32        }
33        for subcommand in &command.subcommands {
34            recursively_add_context_menu_commands(builder, subcommand);
35        }
36    }
37
38    let command_iter = commands.into_iter();
39    let mut commands_builder = Vec::with_capacity(command_iter.size_hint().0);
40    for command in command_iter {
41        if let Some(slash_command) = command.create_as_slash_command() {
42            commands_builder.push(slash_command);
43        }
44        recursively_add_context_menu_commands(&mut commands_builder, command);
45    }
46    commands_builder
47}

49/// Registers the given list of application commands to Discord as global commands.
50///
51/// Thin wrapper around [`create_application_commands`] that funnels the returned builder into
52/// [`serenity::Command::set_global_commands`].
53pub async fn register_globally<'a, U: 'a, E: 'a>(
54    http: impl AsRef<serenity::Http>,
55    commands: impl IntoIterator<Item = &'a crate::Command<U, E>>,
56) -> Result<(), serenity::Error> {
57    let builder = create_application_commands(commands);
58    serenity::Command::set_global_commands(http, builder).await?;
59    Ok(())
60}

62/// Registers the given list of application commands to Discord as guild-specific commands.
63///
64/// Thin wrapper around [`create_application_commands`] that funnels the returned builder into
65/// [`serenity::GuildId::set_commands`].
66pub async fn register_in_guild<'a, U: 'a, E: 'a>(
67    http: impl AsRef<serenity::Http>,
68    commands: impl IntoIterator<Item = &'a crate::Command<U, E>>,
69    guild_id: serenity::GuildId,
70) -> Result<(), serenity::Error> {
71    let builder = create_application_commands(commands);
72    guild_id.set_commands(http, builder).await?;
73    Ok(())
74}

76/// _Note: you probably want [`register_application_commands_buttons`] instead; it's easier and more
77/// powerful_
78///
79/// Wraps [`create_application_commands`] and adds a bot owner permission check and status messages.
80///
81/// This function is supposed to be a ready-to-use implementation for a `~register` command of your
82/// bot. So if you want, you can copy paste this help message for the command:
83///
84/// ```text
85/// Registers application commands in this guild or globally
86///
87/// Run with no arguments to register in guild, run with argument "global" to register globally.
88/// ```
89pub async fn register_application_commands<U, E>(
90    ctx: crate::Context<'_, U, E>,
91    global: bool,
92) -> Result<(), serenity::Error> {
93    let is_bot_owner = ctx.framework().options().owners.contains(&ctx.author().id);
94    if !is_bot_owner {
95        ctx.say("Can only be used by bot owner").await?;
96        return Ok(());
97    }
98
99    let commands_builder = create_application_commands(&ctx.framework().options().commands);
100    let num_commands = commands_builder.len();
101
102    if global {
103        ctx.say(format!("Registering {num_commands} commands...",))
104            .await?;
105        serenity::Command::set_global_commands(ctx, commands_builder).await?;
106    } else {
107        let guild_id = match ctx.guild_id() {
108            Some(x) => x,
109            None => {
110                ctx.say("Must be called in guild").await?;
111                return Ok(());
112            }
113        };
114
115        ctx.say(format!("Registering {num_commands} commands..."))
116            .await?;
117        guild_id.set_commands(ctx, commands_builder).await?;
118    }
119
120    ctx.say("Done!").await?;
121
122    Ok(())
123}

125/// Spawns four buttons to register or delete application commands globally or in the current guild
126///
127/// Upgraded version of [`register_application_commands`]
128///
129/// ![Screenshot of output](https://imgur.com/rTbTaDs.png)
130///
131/// You probably want to use this by wrapping it in a small `register` command:
132/// ```rust
133/// # type Error = Box<dyn std::error::Error + Send + Sync>;
134/// # type Context<'a> = poise::Context<'a, (), Error>;
135/// #[poise::command(prefix_command)]
136/// pub async fn register(ctx: Context<'_>) -> Result<(), Error> {
137///     poise::builtins::register_application_commands_buttons(ctx).await?;
138///     Ok(())
139/// }
140///
141/// // ...
142/// poise::FrameworkOptions {
143///     commands: vec![
144///         // ...
145///         register(),
146///     ],
147/// #   ..Default::default()
148/// };
149/// ```
150///
151/// Which you can call like any prefix command, for example `@your_bot register`.
152pub async fn register_application_commands_buttons<U, E>(
153    ctx: crate::Context<'_, U, E>,
154) -> Result<(), serenity::Error> {
155    let create_commands = create_application_commands(&ctx.framework().options().commands);
156    let num_commands = create_commands.len();
157
158    let is_bot_owner = ctx.framework().options().owners.contains(&ctx.author().id);
159    if !is_bot_owner {
160        ctx.say("Can only be used by bot owner").await?;
161        return Ok(());
162    }
163
164    let components = serenity::CreateActionRow::Buttons(vec![
165        serenity::CreateButton::new("register.guild")
166            .label("Register in guild")
167            .style(serenity::ButtonStyle::Primary)
168            .emoji('📋'),
169        serenity::CreateButton::new("unregister.guild")
170            .label("Delete in guild")
171            .style(serenity::ButtonStyle::Danger)
172            .emoji('🗑'),
173        serenity::CreateButton::new("register.global")
174            .label("Register globally")
175            .style(serenity::ButtonStyle::Primary)
176            .emoji('📋'),
177        serenity::CreateButton::new("unregister.global")
178            .label("Unregister globally")
179            .style(serenity::ButtonStyle::Danger)
180            .emoji('🗑'),
181    ]);
182
183    let builder = crate::CreateReply::default()
184        .content("Choose what to do with the commands:")
185        .components(vec![components]);
186
187    let reply = ctx.send(builder).await?;
188
189    let interaction = reply
190        .message()
191        .await?
192        .await_component_interaction(ctx)
193        .author_id(ctx.author().id)
194        .await;
195
196    reply
197        .edit(
198            ctx,
199            crate::CreateReply::default()
200                .components(vec![])
201                .content("Processing... Please wait."),
202        )
203        .await?; // remove buttons after button press and edit message
204    let pressed_button_id = match &interaction {
205        Some(m) => &m.data.custom_id,
206        None => {
207            ctx.say(":warning: You didn't interact in time - please run the command again.")
208                .await?;
209            return Ok(());
210        }
211    };
212
213    let (register, global) = match &**pressed_button_id {
214        "register.global" => (true, true),
215        "unregister.global" => (false, true),
216        "register.guild" => (true, false),
217        "unregister.guild" => (false, false),
218        other => {
219            tracing::warn!("unknown register button ID: {:?}", other);
220            return Ok(());
221        }
222    };
223
224    let start_time = std::time::Instant::now();
225
226    if global {
227        if register {
228            ctx.say(format!(
229                ":gear: Registering {num_commands} global commands...",
230            ))
231            .await?;
232            serenity::Command::set_global_commands(ctx, create_commands).await?;
233        } else {
234            ctx.say(":gear: Unregistering global commands...").await?;
235            serenity::Command::set_global_commands(ctx, vec![]).await?;
236        }
237    } else {
238        let guild_id = match ctx.guild_id() {
239            Some(x) => x,
240            None => {
241                ctx.say(":x: Must be called in guild").await?;
242                return Ok(());
243            }
244        };
245        if register {
246            ctx.say(format!(
247                ":gear: Registering {num_commands} guild commands...",
248            ))
249            .await?;
250            guild_id.set_commands(ctx, create_commands).await?;
251        } else {
252            ctx.say(":gear: Unregistering guild commands...").await?;
253            guild_id.set_commands(ctx, vec![]).await?;
254        }
255    }
256
257    // Calulate time taken and send message
258    let time_taken = start_time.elapsed();
259    ctx.say(format!(
260        ":white_check_mark: Done! Took {}ms",
261        time_taken.as_millis()
262    ))
263    .await?;
264
265    Ok(())
266}
