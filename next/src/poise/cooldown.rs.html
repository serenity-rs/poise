1//! Infrastructure for command cooldowns
2
3use crate::serenity_prelude as serenity;
4// I usually don't really do imports, but these are very convenient
5use std::collections::HashMap;
6use std::time::{Duration, Instant};
7
8/// Subset of [`crate::Context`] so that [`Cooldowns`] can be used without requiring a full [Context](`crate::Context`)
9/// (ie from within an `event_handler`)
10#[derive(Default, Clone, PartialEq, Eq, Debug, Hash)]
11pub struct CooldownContext {
12    /// The user associated with this request
13    pub user_id: serenity::UserId,
14    /// The guild this request originated from or `None`
15    pub guild_id: Option<serenity::GuildId>,
16    /// The channel associated with this request
17    pub channel_id: serenity::ChannelId,
18}
19
20/// Configuration struct for [`Cooldowns`]
21#[derive(Default, Clone, PartialEq, Eq, Debug, Hash)]
22pub struct CooldownConfig {
23    /// This cooldown operates on a global basis
24    pub global: Option<Duration>,
25    /// This cooldown operates on a per-user basis
26    pub user: Option<Duration>,
27    /// This cooldown operates on a per-guild basis
28    pub guild: Option<Duration>,
29    /// This cooldown operates on a per-channel basis
30    pub channel: Option<Duration>,
31    /// This cooldown operates on a per-member basis
32    pub member: Option<Duration>,
33    #[doc(hidden)]
34    pub __non_exhaustive: (),
35}
36
37/// Tracks all types of cooldowns for a single command
38///
39/// You probably don't need to use this directly. `#[poise::command]` automatically generates a
40/// cooldown handler.
41#[derive(Default, Clone, Debug, PartialEq, Eq)]
42pub struct CooldownTracker {
43    /// Stores the timestamp of the last global invocation
44    global_invocation: Option<Instant>,
45    /// Stores the timestamps of the last invocation per user
46    user_invocations: HashMap<serenity::UserId, Instant>,
47    /// Stores the timestamps of the last invocation per guild
48    guild_invocations: HashMap<serenity::GuildId, Instant>,
49    /// Stores the timestamps of the last invocation per channel
50    channel_invocations: HashMap<serenity::ChannelId, Instant>,
51    /// Stores the timestamps of the last invocation per member (user and guild)
52    member_invocations: HashMap<(serenity::UserId, serenity::GuildId), Instant>,
53}
54
55/// Possible types of command cooldowns.
56///
57/// Currently used for [CooldownTracker::set_last_invocation]
58#[non_exhaustive]
59pub enum CooldownType {
60    /// A global cooldown that applies to all users, channels, and guilds.
61    Global,
62    /// A cooldown specific to individual users.
63    User(serenity::UserId),
64    /// A cooldown that applies to an entire guild.
65    Guild(serenity::GuildId),
66    /// A cooldown specific to individual channels.
67    Channel(serenity::ChannelId),
68    /// A cooldown specific to individual members within a guild.
69    Member((serenity::UserId, serenity::GuildId)),
70}
71
72/// **Renamed to [`CooldownTracker`]**
73pub use CooldownTracker as Cooldowns;
74
75impl CooldownTracker {
76    /// Create a new cooldown tracker
77    pub fn new() -> Self {
78        Self {
79            global_invocation: None,
80            user_invocations: HashMap::new(),
81            guild_invocations: HashMap::new(),
82            channel_invocations: HashMap::new(),
83            member_invocations: HashMap::new(),
84        }
85    }
86
87    /// Queries the cooldown buckets and checks if all cooldowns have expired and command
88    /// execution may proceed. If not, Some is returned with the remaining cooldown
89    pub fn remaining_cooldown(
90        &self,
91        ctx: CooldownContext,
92        cooldown_durations: &CooldownConfig,
93    ) -> Option<Duration> {
94        let mut cooldown_data = vec![
95            (cooldown_durations.global, self.global_invocation),
96            (
97                cooldown_durations.user,
98                self.user_invocations.get(&ctx.user_id).copied(),
99            ),
100            (
101                cooldown_durations.channel,
102                self.channel_invocations.get(&ctx.channel_id).copied(),
103            ),
104        ];
105
106        if let Some(guild_id) = ctx.guild_id {
107            cooldown_data.push((
108                cooldown_durations.guild,
109                self.guild_invocations.get(&guild_id).copied(),
110            ));
111            cooldown_data.push((
112                cooldown_durations.member,
113                self.member_invocations
114                    .get(&(ctx.user_id, guild_id))
115                    .copied(),
116            ));
117        }
118
119        cooldown_data
120            .iter()
121            .filter_map(|&(cooldown, last_invocation)| {
122                let duration_since = Instant::now().saturating_duration_since(last_invocation?);
123                let cooldown_left = cooldown?.checked_sub(duration_since)?;
124                Some(cooldown_left)
125            })
126            .max()
127    }
128
129    /// Indicates that a command has been executed and all associated cooldowns should start running
130    pub fn start_cooldown(&mut self, ctx: CooldownContext) {
131        let now = Instant::now();
132
133        self.global_invocation = Some(now);
134        self.user_invocations.insert(ctx.user_id, now);
135        self.channel_invocations.insert(ctx.channel_id, now);
136
137        if let Some(guild_id) = ctx.guild_id {
138            self.guild_invocations.insert(guild_id, now);
139            self.member_invocations.insert((ctx.user_id, guild_id), now);
140        }
141    }
142
143    /// Sets the last invocation for the specified cooldown bucket.
144    ///
145    /// This function is not usually needed for regular usage. It was added to allow for extra
146    /// flexibility in cases where you might want to shorten or lengthen a cooldown after
147    /// invocation.
148    pub fn set_last_invocation(&mut self, cooldown_type: CooldownType, instant: Instant) {
149        match cooldown_type {
150            CooldownType::Global => self.global_invocation = Some(instant),
151            CooldownType::User(user_id) => {
152                self.user_invocations.insert(user_id, instant);
153            }
154            CooldownType::Guild(guild_id) => {
155                self.guild_invocations.insert(guild_id, instant);
156            }
157            CooldownType::Channel(channel_id) => {
158                self.channel_invocations.insert(channel_id, instant);
159            }
160            CooldownType::Member(member) => {
161                self.member_invocations.insert(member, instant);
162            }
163        }
164    }
165}

167impl<'a> From<&'a serenity::Message> for CooldownContext {
168    fn from(message: &'a serenity::Message) -> Self {
169        Self {
170            user_id: message.author.id,
171            channel_id: message.channel_id,
172            guild_id: message.guild_id,
173        }
174    }
175}
