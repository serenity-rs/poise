3use super::*;
4
5/// A command parameter type for key-value args
6///
7/// For example `key1=value1 key2="value2 with spaces"`
8#[derive(Clone, PartialEq, Eq, Debug, Default)]
9pub struct KeyValueArgs(pub std::collections::HashMap<String, String>);
10
11impl KeyValueArgs {
12    /// Retrieve a single value by its key
13    pub fn get(&self, key: &str) -> Option<&str> {
14        self.0.get(key).map(|x| x.as_str())
15    }
16
17    /// Reads a single key value pair ("key=value") from the front of the arguments
18    fn pop_single_key_value_pair(args: &str) -> Option<(&str, (String, String))> {
19        // TODO: share quote parsing machinery with PopArgument impl for String
20
21        if args.is_empty() {
22            return None;
23        }
24
25        let mut key = String::new();
26        let mut inside_string = false;
27        let mut escaping = false;
28
29        let mut chars = args.trim_start().chars();
30        loop {
31            let c = chars.next()?;
32            if escaping {
33                key.push(c);
34                escaping = false;
35            } else if !inside_string && c.is_whitespace() {
36                return None;
37            } else if c == '"' {
38                inside_string = !inside_string;
39            } else if c == '\\' {
40                escaping = true;
41            } else if !inside_string && c == '=' {
42                break;
43            } else if !inside_string && c.is_ascii_punctuation() {
44                // If not enclosed in quotes, keys mustn't contain special characters.
45                // Otherwise this command invocation: "?eval `0..=5`" is parsed as key-value args
46                // with key "`0.." and value "5`". (This was a long-standing issue in rustbot)
47                return None;
48            } else {
49                key.push(c);
50            }
51        }
52
53        let args = chars.as_str();
54        // `args` used to contain "key=value ...", now it contains "value ...", so pop the value off
55        let (args, value) = super::pop_string(args).unwrap_or((args, String::new()));
56
57        Some((args, (key, value)))
58    }
59
60    /// Reads as many key-value args as possible from the front of the string and produces a
61    /// [`KeyValueArgs`] out of those
62    fn pop_from(mut args: &str) -> (&str, Self) {
63        let mut pairs = std::collections::HashMap::new();
64
65        while let Some((remaining_args, (key, value))) = Self::pop_single_key_value_pair(args) {
66            args = remaining_args;
67            pairs.insert(key, value);
68        }
69
70        (args, Self(pairs))
71    }
72}
73
74#[async_trait::async_trait]
75impl<'a> PopArgument<'a> for KeyValueArgs {
76    async fn pop_from(
77        args: &'a str,
78        attachment_index: usize,
79        _: &serenity::Context,
80        _: &serenity::Message,
81    ) -> Result<(&'a str, usize, Self), (Box<dyn std::error::Error + Send + Sync>, Option<String>)>
82    {
83        let (a, b) = Self::pop_from(args);
84
85        Ok((a, attachment_index, b))
86    }
87}

89#[cfg(test)]
90#[test]
91fn test_key_value_args() {
92    for &(string, pairs, remaining_args) in &[
93        (
94            r#"key1=value1 key2=value2"#,
95            &[("key1", "value1"), ("key2", "value2")][..],
96            "",
97        ),
98        (
99            r#""key 1"=value\ 1 key\ 2="value 2""#,
100            &[("key 1", "value 1"), ("key 2", "value 2")],
101            "",
102        ),
103        (
104            r#"key1"=value1 key2=value2"#,
105            &[],
106            r#"key1"=value1 key2=value2"#,
107        ),
108        (r#"dummyval"#, &[], "dummyval"),
109        (r#"dummyval="#, &[("dummyval", "")], ""),
110    ] {
111        let (args, kv_args) = KeyValueArgs::pop_from(string);
112
113        assert_eq!(
114            kv_args.0,
115            pairs
116                .iter()
117                .map(|&(k, v)| (k.to_owned(), v.to_owned()))
118                .collect(),
119        );
120        assert_eq!(args, remaining_args);
121    }
122}
