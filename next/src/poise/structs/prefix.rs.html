7/// The event that triggered a prefix command execution
8#[derive(Copy, Clone, Debug, PartialEq, Eq)]
9pub enum MessageDispatchTrigger {
10    /// The invocation message was posted directly (common case)
11    MessageCreate,
12    /// The message was edited, and was already a valid invocation pre-edit
13    MessageEdit,
14    /// The message was edited, and was not a valid invocation pre-edit (i.e. user typoed the
15    /// command, then fixed it)
16    MessageEditFromInvalid,
17    #[doc(hidden)]
18    __NonExhaustive,
19}

21/// Prefix-specific context passed to command invocations.
22///
23/// Contains the trigger message, the Discord connection management stuff, and the user data.
24#[derive(derivative::Derivative)]
25#[derivative(Debug(bound = ""))]
26pub struct PrefixContext<'a, U, E> {
27    /// The invoking user message
28    pub msg: &'a serenity::Message,
29    /// Prefix used by the user to invoke this command
30    pub prefix: &'a str,
31    /// Command name used by the user to invoke this command
32    pub invoked_command_name: &'a str,
33    /// Entire argument string
34    pub args: &'a str,
35    /// Read-only reference to the framework
36    ///
37    /// Useful if you need the list of commands, for example for a custom help command
38    #[derivative(Debug = "ignore")]
39    pub framework: crate::FrameworkContext<'a, U, E>,
40    /// If the invoked command was a subcommand, these are the parent commands, ordered top down.
41    pub parent_commands: &'a [&'a crate::Command<U, E>],
42    /// The command object which is the current command
43    pub command: &'a crate::Command<U, E>,
44    /// Custom user data carried across a single command invocation
45    pub invocation_data: &'a tokio::sync::Mutex<Box<dyn std::any::Any + Send + Sync>>,
46    /// How this command invocation was triggered
47    pub trigger: MessageDispatchTrigger,
48    /// The function that is called to execute the actual command
49    #[derivative(Debug = "ignore")]
50    pub action: fn(
51        PrefixContext<'_, U, E>,
52    ) -> crate::BoxFuture<'_, Result<(), crate::FrameworkError<'_, U, E>>>,
53
54    // #[non_exhaustive] forbids struct update syntax for ?? reason
55    #[doc(hidden)]
56    pub __non_exhaustive: (),
57}
58// manual Copy+Clone implementations because Rust is getting confused about the type parameter
59impl<U, E> Clone for PrefixContext<'_, U, E> {
60    fn clone(&self) -> Self {
61        *self
62    }
63}
64impl<U, E> Copy for PrefixContext<'_, U, E> {}
65impl<U, E> crate::_GetGenerics for PrefixContext<'_, U, E> {
66    type U = U;
67    type E = E;
68}

70/// Possible ways to define a command prefix
71#[derive(Clone, Debug)]
72pub enum Prefix {
73    /// A case-sensitive string literal prefix (passed to [`str::strip_prefix`])
74    Literal(&'static str),
75    /// Regular expression which matches the prefix
76    Regex(regex::Regex),
77    #[doc(hidden)]
78    __NonExhaustive,
79}

81/// Prefix-specific framework configuration
82#[derive(derivative::Derivative)]
83#[derivative(Debug(bound = ""))]
84pub struct PrefixFrameworkOptions<U, E> {
85    /// The main bot prefix. Can be set to None if the bot supports only
86    /// [dynamic prefixes](Self::dynamic_prefix).
87    pub prefix: Option<Cow<'static, str>>,
88    /// List of additional bot prefixes
89    // TODO: maybe it would be nicer to have separate fields for literal and regex prefixes
90    // That way, you don't need to wrap every single literal prefix in a long path which looks ugly
91    pub additional_prefixes: Vec<Prefix>,
92    /// Callback invoked on every message to return a prefix.
93    ///
94    /// Override this field for a simple dynamic prefix which changes depending on the guild or user.
95    ///
96    /// For more advanced dynamic prefixes, see [`Self::stripped_dynamic_prefix`]
97    #[derivative(Debug = "ignore")]
98    pub dynamic_prefix: Option<
99        fn(crate::PartialContext<'_, U, E>) -> BoxFuture<'_, Result<Option<Cow<'static, str>>, E>>,
100    >,
101    /// Callback invoked on every message to strip the prefix off an incoming message.
102    ///
103    /// Override this field for advanced dynamic prefixes which change depending on guild or user.
104    ///
105    /// Return value is a tuple of the prefix and the rest of the message:
106    /// ```rust,no_run
107    /// # poise::PrefixFrameworkOptions::<(), ()> { stripped_dynamic_prefix: Some(|_, msg, _| Box::pin(async move {
108    /// let my_cool_prefix = "$";
109    /// if msg.content.starts_with(my_cool_prefix) {
110    ///     return Ok(Some(msg.content.split_at(my_cool_prefix.len())));
111    /// }
112    /// Ok(None)
113    /// # })), ..Default::default() };
114    /// ```
115    #[derivative(Debug = "ignore")]
116    pub stripped_dynamic_prefix: Option<
117        for<'a> fn(
118            &'a serenity::Context,
119            &'a serenity::Message,
120            &'a U,
121        ) -> BoxFuture<'a, Result<Option<(&'a str, &'a str)>, E>>,
122    >,
123    /// Treat a bot mention (a ping) like a prefix
124    pub mention_as_prefix: bool,
125    /// If Some, the framework will react to message edits by editing the corresponding bot response
126    /// with the new result.
127    pub edit_tracker: Option<std::sync::Arc<std::sync::RwLock<crate::EditTracker>>>,
128    /// If the user makes a typo in their message and a subsequent edit creates a valid invocation,
129    /// the bot will execute the command if this attribute is set. [`Self::edit_tracker`] does not
130    /// need to be set for this.
131    ///
132    /// That does not mean that any subsequent edits will also trigger execution. For that,
133    /// see [`crate::Command::invoke_on_edit`].
134    ///
135    /// Note: only has an effect if [`Self::edit_tracker`] is set.
136    pub execute_untracked_edits: bool,
137    /// Whether to ignore message edits on messages that have not yet been responded to.
138    ///
139    /// This is the case if the message edit happens before a command has sent a response, or if the
140    /// command does not send a response at all.
141    pub ignore_edits_if_not_yet_responded: bool,
142
143    /// Whether commands in messages emitted by this bot itself should be executed as well.
144    pub execute_self_messages: bool,
145    /// Whether to ignore messages from bots for command invoking. Default `true`
146    pub ignore_bots: bool,
147    /// Whether to ignore commands contained within thread creation messages. Default `true`
148    pub ignore_thread_creation: bool,
149    /// Whether command names should be compared case-insensitively.
150    pub case_insensitive_commands: bool,
151    /// Callback for all non-command messages. Useful if you want to run code on any message that
152    /// is not a command
153    pub non_command_message: Option<
154        for<'a> fn(
155            &'a crate::FrameworkContext<'a, U, E>,
156            &'a serenity::Message,
157        ) -> crate::BoxFuture<'a, Result<(), E>>,
158    >,
159    /* // TODO: implement
160    /// Whether to invoke help command when someone sends a message with just a bot mention
161    pub help_when_mentioned: bool,
162    /// The bot's general help command. Currently used for [`Self::help_when_mentioned`].
163    pub help_commmand: Option<Command<U, E>>,
164    // /// The bot's help command for individial commands. Currently used when a command group without
165    // /// any specific subcommand is invoked. This command is expected to take the command name as a
166    // /// single parameter
167    // pub command_specific_help_commmand: Option<Command<U, E>>, */
168    // #[non_exhaustive] forbids struct update syntax for ?? reason
169    #[doc(hidden)]
170    pub __non_exhaustive: (),
171}
