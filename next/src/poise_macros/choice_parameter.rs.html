1//! Implements the #[derive(ChoiceParameter)] derive macro
2
3use proc_macro::TokenStream;
4use syn::spanned::Spanned as _;
5
6/// Representation of the function parameter attribute arguments
7#[derive(Debug, darling::FromMeta)]
8#[darling(allow_unknown_fields)]
9struct VariantAttribute {
10    // Note to self: when adding an attribute here, add it to #[proc_macro_derive]!
11    #[darling(multiple)]
12    name: Vec<String>,
13    #[darling(multiple)]
14    name_localized: Vec<crate::util::Tuple2<String>>,
15}
16
17pub fn choice_parameter(input: syn::DeriveInput) -> Result<TokenStream, darling::Error> {
18    let enum_ = match input.data {
19        syn::Data::Enum(x) => x,
20        _ => {
21            return Err(syn::Error::new(
22                input.ident.span(),
23                "Only enums can be used for choice parameters",
24            )
25            .into())
26        }
27    };
28
29    let mut variant_idents: Vec<proc_macro2::Ident> = Vec::new();
30    let mut names: Vec<String> = Vec::new();
31    let mut alternative_names = Vec::new();
32    let mut locales: Vec<Vec<String>> = Vec::new();
33    let mut localized_names: Vec<Vec<String>> = Vec::new();
34
35    for variant in enum_.variants {
36        if !matches!(&variant.fields, syn::Fields::Unit) {
37            return Err(syn::Error::new(
38                variant.fields.span(),
39                "Choice parameters cannot have fields",
40            )
41            .into());
42        }
43
44        let attrs: Vec<_> = variant
45            .attrs
46            .into_iter()
47            .map(|attr| darling::ast::NestedMeta::Meta(attr.meta))
48            .collect();
49        let mut attrs = <VariantAttribute as darling::FromMeta>::from_list(&attrs)?;
50
51        let main_name = if attrs.name.is_empty() {
52            variant.ident.to_string()
53        } else {
54            attrs.name.remove(0)
55        };
56
57        variant_idents.push(variant.ident);
58        names.push(main_name);
59        alternative_names.push(attrs.name);
60
61        let (a, b) = attrs.name_localized.into_iter().map(|x| (x.0, x.1)).unzip();
62        locales.push(a);
63        localized_names.push(b);
64    }
65
66    let enum_ident = &input.ident;
67    let indices = 0..variant_idents.len();
68    Ok(quote::quote! {
69        impl poise::ChoiceParameter for #enum_ident {
70            fn list() -> std::borrow::Cow<'static, [poise::CommandParameterChoice]> {
71                use ::std::borrow::Cow;
72
73                Cow::Borrowed(&[ #( poise::CommandParameterChoice {
74                    __non_exhaustive: (),
75                    name: Cow::Borrowed(#names),
76                    localizations: Cow::Borrowed(&[
77                        #( (Cow::Borrowed(#locales), Cow::Borrowed(#localized_names)) ),*
78                    ]),
79                }, )* ])
80            }
81
82            fn from_index(index: usize) -> Option<Self> {
83                match index {
84                    #( #indices => Some(Self::#variant_idents), )*
85                    _ => None,
86                }
87            }
88
89            fn from_name(name: &str) -> Option<Self> {
90                #( if name.eq_ignore_ascii_case(#names)
91                    #( || name.eq_ignore_ascii_case(#alternative_names) )*
92                {
93                    return Some(Self::#variant_idents);
94                } )*
95                None
96            }
97
98            fn name(&self) -> &'static str {
99                match self {
100                    #( Self::#variant_idents => #names, )*
101                }
102            }
103
104            fn localized_name(&self, locale: &str) -> Option<&'static str> {
105                match self {
106                    #( Self::#variant_idents => match locale {
107                        #( #locales => Some(#localized_names), )*
108                        _ => None,
109                    }, )*
110                }
111            }
112        }
113    }
114    .into())
115}
