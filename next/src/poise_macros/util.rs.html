2// transforms a type of form `OuterType<T>` into `T`
3pub fn extract_type_parameter<'a>(outer_type: &str, t: &'a syn::Type) -> Option<&'a syn::Type> {
4    if let syn::Type::Path(path) = t {
5        if path.path.segments.len() == 1 {
6            let path = &path.path.segments[0];
7            if path.ident == outer_type {
8                if let syn::PathArguments::AngleBracketed(generics) = &path.arguments {
9                    if generics.args.len() == 1 {
10                        if let syn::GenericArgument::Type(t) = &generics.args[0] {
11                            return Some(t);
12                        }
13                    }
14                }
15            }
16        }
17    }
18    None
19}

21/// Converts None => `None` and Some(x) => `Some(#x)`
22pub fn wrap_option<T: quote::ToTokens>(literal: Option<T>) -> syn::Expr {
23    match literal {
24        Some(literal) => syn::parse_quote! { Some(#literal) },
25        None => syn::parse_quote! { None },
26    }
27}

29/// Converts None => `None` and Some(x) => `Some(#map_path(#x))`
30pub fn wrap_option_and_map<T: quote::ToTokens>(
31    literal: Option<T>,
32    map_path: impl quote::ToTokens,
33) -> syn::Expr {
34    match literal {
35        Some(literal) => syn::parse_quote! { Some(#map_path(#literal)) },
36        None => syn::parse_quote! { None },
37    }
38}

40pub fn wrap_option_to_string<T: quote::ToTokens>(literal: Option<T>) -> syn::Expr {
41    let cowstr_path = quote::quote!(Cow::Borrowed);
42    wrap_option_and_map(literal, cowstr_path)
43}

54/// Darling utility type that accepts a list of things, e.g. `#[attr(thing1, thing2...)]`
55#[derive(Debug)]
56pub struct List<T>(pub Vec<T>);
57impl<T: darling::FromMeta> darling::FromMeta for List<T> {
58    fn from_list(items: &[::darling::ast::NestedMeta]) -> darling::Result<Self> {
59        items
60            .iter()
61            .map(|item| T::from_nested_meta(item))
62            .collect::<darling::Result<Vec<T>>>()
63            .map(Self)
64    }
65}
66impl<T> Default for List<T> {
67    fn default() -> Self {
68        Self(Default::default())
69    }
70}

72/// Darling utility type that accepts a 2-tuple list of things, e.g. `#[attr(thing1, thing2)]`
73#[derive(Debug)]
74pub struct Tuple2<T>(pub T, pub T);
75impl<T: darling::FromMeta> darling::FromMeta for Tuple2<T> {
76    fn from_list(items: &[::darling::ast::NestedMeta]) -> darling::Result<Self> {
77        Ok(match items {
78            [a, b] => Self(T::from_nested_meta(a)?, T::from_nested_meta(b)?),
79            _ => {
80                return Err(syn::Error::new(
81                    proc_macro2::Span::call_site(),
82                    "expected two items `(\"a\", \"b\")`",
83                )
84                .into())
85            }
86        })
87    }
88}

102pub fn iter_tuple_2_to_vec_map<I, T>(v: I) -> proc_macro2::TokenStream
103where
104    I: ExactSizeIterator<Item = Tuple2<T>>,
105    T: quote::ToTokens,
106{
107    if v.len() == 0 {
108        return quote::quote!(Cow::Borrowed(&[]));
109    }
110
111    let (keys, values) = v
112        .into_iter()
113        .map(|x| (x.0, x.1))
114        .unzip::<_, _, Vec<_>, Vec<_>>();
115
116    quote::quote! {
117        Cow::Borrowed(&[
118            #( (Cow::Borrowed(#keys), Cow::Borrowed(#values)) ),*
119        ])
120    }
