25pub async fn component_modal(ctx: crate::Context<'_>) -> Result<(), Error> {
26    let reply = {
27        let components = vec![serenity::CreateActionRow::Buttons(vec![
28            serenity::CreateButton::new("open_modal")
29                .label("Open modal")
30                .style(poise::serenity_prelude::ButtonStyle::Success),
31        ])];
32
33        poise::CreateReply::default()
34            .content("Click the button below to open the modal")
35            .components(components)
36    };
37
38    ctx.send(reply).await?;
39
40    let serenity_ctx = ctx.serenity_context();
41    while let Some(mci) = serenity::ComponentInteractionCollector::new(serenity_ctx)
42        .timeout(std::time::Duration::from_secs(120))
43        .filter(move |mci| mci.data.custom_id == "open_modal")
44        .await
45    {
46        let data =
47            poise::execute_modal_on_component_interaction::<MyModal>(serenity_ctx, mci, None, None)
48                .await?;
49        println!("Got data: {:?}", data);
50    }
51    Ok(())
52}

6pub async fn boop(ctx: Context<'_>) -> Result<(), Error> {
7    let uuid_boop = ctx.id();
8
9    let reply = {
10        let components = vec![serenity::CreateActionRow::Buttons(vec![
11            serenity::CreateButton::new(format!("{uuid_boop}"))
12                .style(serenity::ButtonStyle::Primary)
13                .label("Boop me!"),
14        ])];
15
16        CreateReply::default()
17            .content("I want some boops!")
18            .components(components)
19    };
20
21    ctx.send(reply).await?;
22
23    let mut boop_count = 0;
24    while let Some(mci) = serenity::ComponentInteractionCollector::new(ctx)
25        .author_id(ctx.author().id)
26        .channel_id(ctx.channel_id())
27        .timeout(std::time::Duration::from_secs(120))
28        .filter(move |mci| mci.data.custom_id == uuid_boop.to_string())
29        .await
30    {
31        boop_count += 1;
32
33        let mut msg = mci.message.clone();
34        msg.edit(
35            ctx,
36            serenity::EditMessage::new().content(format!("Boop count: {boop_count}")),
37        )
38        .await?;
39
40        mci.create_response(ctx, serenity::CreateInteractionResponse::Acknowledge)
41            .await?;
42    }
43
44    Ok(())
45}

pub fn reply(self, reply: bool) -> Self

Makes this message an inline reply to another message like serenity::Message::reply (prefix-only, because slash commands are always inline replies anyways).

To disable the ping, set Self::allowed_mentions with serenity::CreateAllowedMentions::replied_user set to false.
