1use poise::serenity_prelude as serenity;
2
3struct Data {} // User data, which is stored and accessible in all command invocations
4type Error = Box<dyn std::error::Error + Send + Sync>;
5type Context<'a> = poise::Context<'a, Data, Error>;
6
7/// Displays your or another user's account creation date
8#[poise::command(slash_command, prefix_command)]
9async fn age(
10    ctx: Context<'_>,
11    #[description = "Selected user"] user: Option<serenity::User>,
12) -> Result<(), Error> {
13    let u = user.as_ref().unwrap_or_else(|| ctx.author());
14    let response = format!("{}'s account was created at {}", u.name, u.created_at());
15    ctx.say(response).await?;
16    Ok(())
17}
18
19#[tokio::main]
20async fn main() {
21    let token = std::env::var("DISCORD_TOKEN").expect("missing DISCORD_TOKEN");
22    let intents = serenity::GatewayIntents::non_privileged();
23
24    let framework = poise::Framework::builder()
25        .options(poise::FrameworkOptions {
26            commands: vec![age()],
27            ..Default::default()
28        })
29        .setup(|ctx, _ready, framework| {
30            Box::pin(async move {
31                poise::builtins::register_globally(ctx, &framework.options().commands).await?;
32                Ok(Data {})
33            })
34        })
35        .build();
36
37    let client = serenity::ClientBuilder::new(token, intents)
38        .framework(framework)
39        .await;
40    client.unwrap().start().await.unwrap();
41}
