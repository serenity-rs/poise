37/// Welcome a user
38#[poise::command(
39    slash_command,
40    name_localized("de", "begrüßen"),
41    name_localized("es-ES", "saludar"),
42    description_localized("de", "Einen Nutzer begrüßen"),
43    description_localized("es-ES", "Saludar a un usuario")
44)]
45pub async fn welcome(
46    ctx: Context<'_>,
47    #[name_localized("de", "nutzer")]
48    #[description_localized("de", "Der zu begrüßende Nutzer")]
49    #[name_localized("es-ES", "usuario")]
50    #[description_localized("es-ES", "El usuario a saludar")]
51    #[description = "The user to welcome"]
52    user: serenity::User,
53    #[name_localized("de", "nachricht")]
54    #[description_localized("de", "Die versendete Nachricht")]
55    #[name_localized("es-ES", "mensaje")]
56    #[description_localized("es-ES", "El mensaje enviado")]
57    #[description = "The message to send"]
58    message: WelcomeChoice,
59) -> Result<(), Error> {
60    let message = message
61        .localized_name(ctx.locale().unwrap_or(""))
62        .unwrap_or_else(|| message.name());
63    ctx.say(format!("<@{}> {}", user.id, message)).await?;
64    Ok(())
65}
