25type Error = Box<dyn std::error::Error + Send + Sync>;
26type Context<'a> = poise::Context<'a, Data, Error>;
27// User data, which is stored and accessible in all command invocations
28pub struct Data {}

30#[tokio::main]
31async fn main() {
32    tracing_subscriber::fmt::init();
33
34    let framework = poise::Framework::builder()

36            commands: vec![
37                attachment_parameter::file_details(),

55                choice_parameter::inline_choice_int(),
56                code_block_parameter::code(),
57                collector::boop(),
58                context_menu::user_info(),

82            ],
83            prefix_options: poise::PrefixFrameworkOptions {
84                prefix: Some("~".into()),
85                non_command_message: Some(|_, msg| {
86                    Box::pin(async move {
87                        println!("non command message!: {}", msg.content);
88                        Ok(())
89                    })
90                }),
91                ..Default::default()
92            },
93            on_error: |error| {
94                Box::pin(async move {
95                    println!("what the hell");
96                    match error {
97                        poise::FrameworkError::ArgumentParse { error, .. } => {
98                            if let Some(error) = error.downcast_ref::<serenity::RoleParseError>() {
99                                println!("Found a RoleParseError: {:?}", error);
100                            } else {
101                                println!("Not a RoleParseError :(");
102                            }
103                        }
104                        other => poise::builtins::on_error(other).await.unwrap(),
105                    }
106                })
107            },
108            ..Default::default()
109        })
110        .setup(move |ctx, _ready, framework| {
111            Box::pin(async move {
112                poise::builtins::register_globally(ctx, &framework.options().commands).await?;
113                Ok(Data {})
114            })
115        })
116        .build();
117
118    let token = std::env::var("DISCORD_TOKEN").unwrap();
119    let intents =
120        serenity::GatewayIntents::non_privileged() | serenity::GatewayIntents::MESSAGE_CONTENT;
121
122    let client = serenity::ClientBuilder::new(token, intents)
123        .framework(framework)
124        .await;
125
126    client.unwrap().start().await.unwrap()
127}
