3use crate::{serenity_prelude as serenity, BoxFuture};
4
5/// Framework configuration
6#[derive(derivative::Derivative)]
7#[derivative(Debug(bound = ""))]
8pub struct FrameworkOptions<U, E> {
9    /// List of commands in the framework
10    pub commands: Vec<crate::Command<U, E>>,
11    /// Provide a callback to be invoked when any user code yields an error.
12    #[derivative(Debug = "ignore")]
13    pub on_error: fn(crate::FrameworkError<'_, U, E>) -> BoxFuture<'_, ()>,
14    /// Called before every command
15    #[derivative(Debug = "ignore")]
16    pub pre_command: fn(crate::Context<'_, U, E>) -> BoxFuture<'_, ()>,
17    /// Called after every command if it was successful (returned Ok)
18    #[derivative(Debug = "ignore")]
19    pub post_command: fn(crate::Context<'_, U, E>) -> BoxFuture<'_, ()>,
20    /// Provide a callback to be invoked before every command. The command will only be executed
21    /// if the callback returns true.
22    ///
23    /// If individual commands add their own check, both callbacks are run and must return true.
24    #[derivative(Debug = "ignore")]
25    pub command_check: Option<fn(crate::Context<'_, U, E>) -> BoxFuture<'_, Result<bool, E>>>,
26    /// If set to true, skips command checks if command was issued by [`FrameworkOptions::owners`]
27    pub skip_checks_for_owners: bool,
28    /// Default set of allowed mentions to use for all responses
29    ///
30    /// By default, user pings are allowed and role pings and everyone pings are filtered
31    pub allowed_mentions: Option<serenity::CreateAllowedMentions>,
32    /// Invoked before every message sent using [`crate::Context::say`] or [`crate::Context::send`]
33    ///
34    /// Allows you to modify every outgoing message in a central place
35    #[derivative(Debug = "ignore")]
36    pub reply_callback:
37        Option<fn(crate::Context<'_, U, E>, crate::CreateReply) -> crate::CreateReply>,
38    /// If `true`, disables automatic cooldown handling before every command invocation.
39    ///
40    /// Useful for implementing custom cooldown behavior. See [`crate::Command::cooldowns`] and
41    /// the methods on [`crate::Cooldowns`] for how to do that.
42    pub manual_cooldowns: bool,
43    /// If `true`, changes behavior of guild_only command check to abort execution if the guild is
44    /// not in cache.
45    ///
46    /// **If `cache` feature is disabled, this has no effect!**
47    pub require_cache_for_guild_check: bool,
48    /// Called on every Discord event. Can be used to react to non-command events, like messages
49    /// deletions or guild updates.
50    #[derivative(Debug = "ignore")]
51    pub event_handler: for<'a> fn(
52        &'a serenity::Context,
53        &'a serenity::FullEvent,
54        crate::FrameworkContext<'a, U, E>,
55        // TODO: redundant with framework
56        &'a U,
57    ) -> BoxFuture<'a, Result<(), E>>,
58    /// Renamed to [`Self::event_handler`]!
59    #[deprecated = "renamed to event_handler"]
60    pub listener: (),
61    /// Prefix command specific options.
62    pub prefix_options: crate::PrefixFrameworkOptions<U, E>,
63    /// User IDs which are allowed to use owners_only commands
64    pub owners: std::collections::HashSet<serenity::UserId>,
65    /// If true, [`Self::owners`] is automatically initialized with the results of
66    /// [`serenity::Http::get_current_application_info()`].
67    ///
68    /// True by default.
69    pub initialize_owners: bool,
70    /// If set and [`Self::initialize_owners`] is `true`, the selected teams will be initialized
71    /// using the results of [`serenity::Http::get_current_application_info()`].
72    ///
73    /// When set to `None`, only users with the Developer and Admin roles are initialized.
74    ///
75    /// None by default.
76    pub initialized_team_roles: Option<Vec<serenity::TeamMemberRole>>,
77    // #[non_exhaustive] forbids struct update syntax for ?? reason
78    #[doc(hidden)]
79    pub __non_exhaustive: (),
80}
81
82impl<U, E> FrameworkOptions<U, E> {
83    /// Add a new command to the framework
84    #[deprecated = "supply commands in FrameworkOptions directly with `commands: vec![...]`"]
85    pub fn command(
86        &mut self,
87        mut command: crate::Command<U, E>,
88        meta_builder: impl FnOnce(&mut crate::Command<U, E>) -> &mut crate::Command<U, E> + 'static,
89    ) {
90        meta_builder(&mut command);
91        self.commands.push(command);
92    }
93}
94
95impl<U, E> Default for FrameworkOptions<U, E>
96where
97    U: Send + Sync,
98    E: std::fmt::Display + std::fmt::Debug + Send,
99{
100    fn default() -> Self {
101        #[allow(deprecated)] // we need to set the listener field
102        Self {
103            commands: Vec::new(),
104            on_error: |error| {
105                Box::pin(async move {
106                    if let Err(e) = crate::builtins::on_error(error).await {
107                        tracing::error!("Error while handling error: {}", e);
108                    }
109                })
110            },
111            event_handler: |_, _, _, _| Box::pin(async { Ok(()) }),
112            listener: (),
113            pre_command: |_| Box::pin(async {}),
114            post_command: |_| Box::pin(async {}),
115            command_check: None,
116            skip_checks_for_owners: false,
117            allowed_mentions: Some(
118                // Only support direct user pings by default
119                serenity::CreateAllowedMentions::default()
120                    .all_users(true)
121                    // https://github.com/serenity-rs/poise/issues/176
122                    .replied_user(true),
123            ),
124            reply_callback: None,
125            manual_cooldowns: false,
126            require_cache_for_guild_check: false,
127            prefix_options: Default::default(),
128            owners: Default::default(),
129            initialize_owners: true,
130            initialized_team_roles: None,
131            __non_exhaustive: (),
132        }
133    }
134}
