1//! A builder struct that allows easy and readable creation of a [`crate::Framework`]
2
3use crate::serenity_prelude as serenity;
4use crate::BoxFuture;
5
6/// A builder to configure a framework.
7///
8/// If one of the following required values is missing, the builder will panic on start:
9/// - [`Self::setup`]
10/// - [`Self::options`]
11pub struct FrameworkBuilder<U, E> {
12    /// Callback for startup code and user data creation
13    setup: Option<
14        Box<
15            dyn Send
16                + Sync
17                + for<'a> FnOnce(
18                    &'a serenity::Context,
19                    &'a serenity::Ready,
20                    &'a crate::Framework<U, E>,
21                ) -> BoxFuture<'a, Result<U, E>>,
22        >,
23    >,
24    /// Framework options
25    options: Option<crate::FrameworkOptions<U, E>>,
26    /// List of framework commands
27    commands: Vec<crate::Command<U, E>>,
28    /// See [`Self::initialize_owners()`]
29    initialize_owners: bool,
30}
31
32impl<U, E> Default for FrameworkBuilder<U, E> {
33    fn default() -> Self {
34        Self {
35            setup: Default::default(),
36            options: Default::default(),
37            commands: Default::default(),
38            initialize_owners: true,
39        }
40    }
41}
42
43impl<U, E> FrameworkBuilder<U, E> {
44    /// Sets the setup callback which also returns the user data struct.
45    #[must_use]
46    pub fn setup<F>(mut self, setup: F) -> Self
47    where
48        F: Send
49            + Sync
50            + 'static
51            + for<'a> FnOnce(
52                &'a serenity::Context,
53                &'a serenity::Ready,
54                &'a crate::Framework<U, E>,
55            ) -> BoxFuture<'a, Result<U, E>>,
56    {
57        self.setup = Some(Box::new(setup) as _);
58        self
59    }
60
61    /// Configure framework options
62    #[must_use]
63    pub fn options(mut self, options: crate::FrameworkOptions<U, E>) -> Self {
64        self.options = Some(options);
65        self
66    }
67
68    /// Whether to add this bot application's owner and team members to
69    /// [`crate::FrameworkOptions::owners`] automatically
70    ///
71    /// `true` by default
72    pub fn initialize_owners(mut self, initialize_owners: bool) -> Self {
73        self.initialize_owners = initialize_owners;
74        self
75    }
76
77    /// Build the framework with the specified configuration.
78    ///
79    /// For more information, see [`FrameworkBuilder`]
80    pub fn build(self) -> crate::Framework<U, E>
81    where
82        U: Send + Sync + 'static,
83        E: Send + 'static,
84    {
85        let setup = self
86            .setup
87            .expect("No user data setup function was provided to the framework");
88        let mut options = self.options.expect("No framework options provided");
89
90        // Build framework options by concatenating user-set options with commands and owners
91        options.commands.extend(self.commands);
92        options.initialize_owners = self.initialize_owners;
93
94        // Create framework with specified settings
95        crate::Framework::new(options, setup)
96    }
97}
