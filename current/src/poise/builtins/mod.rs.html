22/// An error handler that logs errors either via the [`tracing`] crate or via a Discord message. Set
23/// up a logger like tracing subscriber
24/// (e.g. `tracing_subscriber::fmt::init()`) to see the logged errors from this method.
25///
26/// If the user invoked the command wrong ([`crate::FrameworkError::ArgumentParse`]), the command
27/// help is displayed and the user is directed to the help menu.
28///
29/// Can return an error if sending the Discord error message failed. You can decide for yourself
30/// how to handle this, for example:
31/// ```rust,no_run
32/// # async { let error: poise::FrameworkError<'_, (), &str> = todo!();
33/// if let Err(e) = poise::builtins::on_error(error).await {
34///     tracing::error!("Fatal error while sending error message: {}", e);
35/// }
36/// # };
37/// ```
38pub async fn on_error<U, E: std::fmt::Display + std::fmt::Debug>(
39    error: crate::FrameworkError<'_, U, E>,
40) -> Result<(), serenity::Error> {
41    match error {
42        crate::FrameworkError::Setup { error, .. } => {
43            eprintln!("Error in user data setup: {}", error);
44        }
45        crate::FrameworkError::EventHandler { error, event, .. } => tracing::error!(
46            "User event event handler encountered an error on {} event: {}",
47            event.snake_case_name(),
48            error
49        ),
50        crate::FrameworkError::Command { ctx, error } => {
51            let error = error.to_string();
52            eprintln!("An error occured in a command: {}", error);
53
54            let mentions = CreateAllowedMentions::new()
55                .everyone(false)
56                .all_roles(false)
57                .all_users(false);
58
59            ctx.send(
60                CreateReply::default()
61                    .content(error)
62                    .allowed_mentions(mentions),
63            )
64            .await?;
65        }
66        crate::FrameworkError::SubcommandRequired { ctx } => {
67            let subcommands = ctx
68                .command()
69                .subcommands
70                .iter()
71                .map(|s| &*s.name)
72                .collect::<Vec<_>>();
73            let response = format!(
74                "You must specify one of the following subcommands: {}",
75                subcommands.join(", ")
76            );
77            ctx.send(CreateReply::default().content(response).ephemeral(true))
78                .await?;
79        }
80        crate::FrameworkError::CommandPanic { ctx, payload: _ } => {
81            // Not showing the payload to the user because it may contain sensitive info
82            let embed = serenity::CreateEmbed::default()
83                .title("Internal error")
84                .color((255, 0, 0))
85                .description("An unexpected internal error has occurred.");
86
87            ctx.send(CreateReply::default().embed(embed).ephemeral(true))
88                .await?;
89        }
90        crate::FrameworkError::ArgumentParse { ctx, input, error } => {
91            // If we caught an argument parse error, give a helpful error message with the
92            // command explanation if available
93            let usage = match &ctx.command().help_text {
94                Some(help_text) => &**help_text,
95                None => "Please check the help menu for usage information",
96            };
97            let response = if let Some(input) = input {
98                format!(
99                    "**Cannot parse `{}` as argument: {}**\n{}",
100                    input, error, usage
101                )
102            } else {
103                format!("**{}**\n{}", error, usage)
104            };
105
106            let mentions = CreateAllowedMentions::new()
107                .everyone(false)
108                .all_roles(false)
109                .all_users(false);
110
111            ctx.send(
112                CreateReply::default()
113                    .content(response)
114                    .allowed_mentions(mentions),
115            )
116            .await?;
117        }
118        crate::FrameworkError::CommandStructureMismatch { ctx, description } => {
119            tracing::error!(
120                "Error: failed to deserialize interaction arguments for `/{}`: {}",
121                ctx.command.name,
122                description,
123            );
124        }
125        crate::FrameworkError::CommandCheckFailed { ctx, error } => {
126            tracing::error!(
127                "A command check failed in command {} for user {}: {:?}",
128                ctx.command().name,
129                ctx.author().name,
130                error,
131            );
132        }
133        crate::FrameworkError::CooldownHit {
134            remaining_cooldown,
135            ctx,
136        } => {
137            let msg = format!(
138                "You're too fast. Please wait {} seconds before retrying",
139                remaining_cooldown.as_secs()
140            );
141            ctx.send(CreateReply::default().content(msg).ephemeral(true))
142                .await?;
143        }
144        crate::FrameworkError::MissingBotPermissions {
145            missing_permissions,
146            ctx,
147        } => {
148            let msg = format!(
149                "Command cannot be executed because the bot is lacking permissions: {}",
150                missing_permissions,
151            );
152            ctx.send(CreateReply::default().content(msg).ephemeral(true))
153                .await?;
154        }
155        crate::FrameworkError::MissingUserPermissions {
156            missing_permissions,
157            ctx,
158        } => {
159            let response = if let Some(missing_permissions) = missing_permissions {
160                format!(
161                    "You're lacking permissions for `{}{}`: {}",
162                    ctx.prefix(),
163                    ctx.command().name,
164                    missing_permissions,
165                )
166            } else {
167                format!(
168                    "You may be lacking permissions for `{}{}`. Not executing for safety",
169                    ctx.prefix(),
170                    ctx.command().name,
171                )
172            };
173            ctx.send(CreateReply::default().content(response).ephemeral(true))
174                .await?;
175        }
176        crate::FrameworkError::NotAnOwner { ctx } => {
177            let response = "Only bot owners can call this command";
178            ctx.send(CreateReply::default().content(response).ephemeral(true))
179                .await?;
180        }
181        crate::FrameworkError::GuildOnly { ctx } => {
182            let response = "You cannot run this command in DMs.";
183            ctx.send(CreateReply::default().content(response).ephemeral(true))
184                .await?;
185        }
186        crate::FrameworkError::DmOnly { ctx } => {
187            let response = "You cannot run this command outside DMs.";
188            ctx.send(CreateReply::default().content(response).ephemeral(true))
189                .await?;
190        }
191        crate::FrameworkError::NsfwOnly { ctx } => {
192            let response = "You cannot run this command outside NSFW channels.";
193            ctx.send(CreateReply::default().content(response).ephemeral(true))
194                .await?;
195        }
196        crate::FrameworkError::DynamicPrefix { error, msg, .. } => {
197            tracing::error!(
198                "Dynamic prefix failed for message {:?}: {}",
199                msg.content,
200                error
201            );
202        }
203        crate::FrameworkError::UnknownCommand {
204            msg_content,
205            prefix,
206            ..
207        } => {
208            tracing::warn!(
209                "Recognized prefix `{}`, but didn't recognize command name in `{}`",
210                prefix,
211                msg_content,
212            );
213        }
214        crate::FrameworkError::UnknownInteraction { interaction, .. } => {
215            tracing::warn!("received unknown interaction \"{}\"", interaction.data.name);
216        }
217        crate::FrameworkError::NonCommandMessage { error, .. } => {
218            tracing::warn!("error in non-command message handler: {}", error);
219        }
220        crate::FrameworkError::__NonExhaustive(unreachable) => match unreachable {},
221    }
222
223    Ok(())
224}

226/// An autocomplete function that can be used for the command parameter in your help function.
227///
228/// See `examples/feature_showcase` for an example
229#[allow(clippy::unused_async)] // Required for the return type
230pub async fn autocomplete_command<'a, U, E>(
231    ctx: crate::Context<'a, U, E>,
232    partial: &'a str,
233) -> impl Iterator<Item = String> + 'a {
234    ctx.framework()
235        .options()
236        .commands
237        .iter()
238        .filter(move |cmd| cmd.name.starts_with(partial))
239        .map(|cmd| cmd.name.to_string())
240}

242/// Lists servers of which the bot is a member of, including their member counts, sorted
243/// descendingly by member count.
244///
245/// Non-[public](https://support.discord.com/hc/en-us/articles/360030843331-Enabling-Server-Discovery)
246/// guilds are hidden to preserve privacy. When the command is invoked by the bot
247/// owner as an application command, the response will be made ephemeral and private guilds are
248/// unhidden.
249///
250/// Example:
251/// > I am currently in three servers!
252/// > - **A public server** (7123 members)
253/// > - [3 private servers with 456 members total]
254#[cfg(feature = "cache")]
255pub async fn servers<U, E>(ctx: crate::Context<'_, U, E>) -> Result<(), serenity::Error> {
256    use std::fmt::Write as _;
257
258    let show_private_guilds = ctx.framework().options().owners.contains(&ctx.author().id);
259
260    // Aggregate all guilds and sort them by size
261    let mut hidden_guilds = 0;
262    let mut hidden_guilds_members = 0;
263    let mut shown_guilds = Vec::<(String, u64)>::new();
264    for guild_id in ctx.cache().guilds() {
265        match ctx.cache().guild(guild_id) {
266            Some(guild) => {
267                let is_public = guild.features.iter().any(|x| x == "DISCOVERABLE");
268                if !is_public && !show_private_guilds {
269                    hidden_guilds += 1; // private guild whose name and size shouldn't be exposed
270                } else {
271                    shown_guilds.push((guild.name.clone(), guild.member_count))
272                }
273            }
274            None => hidden_guilds += 1, // uncached guild
275        }
276    }
277    shown_guilds.sort_by_key(|(_, member)| u64::MAX - member); // sort largest guilds first
278
279    // Iterate guilds and build up the response message line by line
280    let mut response = format!(
281        "I am currently in {} servers!\n",
282        shown_guilds.len() + hidden_guilds
283    );
284    if show_private_guilds {
285        response.insert_str(0, "_Showing private guilds because you are a bot owner_\n");
286    }
287    let mut guilds = shown_guilds.into_iter().peekable();
288    while let Some((name, member_count)) = guilds.peek() {
289        let line = format!("- **{}** ({} members)\n", name, member_count);
290
291        // Make sure we don't exceed a certain number of characters below the 2000 char limit so
292        // we have enough space for the remaining servers line
293        if response.len() + line.len() > 1940 {
294            for (_remaining_guild_name, members) in guilds {
295                hidden_guilds += 1;
296                hidden_guilds_members += members;
297            }
298            break;
299        }
300
301        response += &line;
302        guilds.next(); // advance peekable iterator
303    }
304    if hidden_guilds > 0 {
305        let _ = writeln!(
306            response,
307            "- {} remaining servers with {} members total",
308            hidden_guilds, hidden_guilds_members
309        );
310    }
311
312    // Final safe guard (shouldn't be hit at the time of writing)
313    if response.len() > 2000 {
314        let mut truncate_at = 2000;
315        while !response.is_char_boundary(truncate_at) {
316            truncate_at -= 1;
317        }
318        response.truncate(truncate_at);
319    }
320
321    // If we show sensitive data (private guilds), it mustn't be made public, so it's ephemeral
322    let reply = CreateReply::default()
323        .content(response)
324        .ephemeral(show_private_guilds);
325
326    ctx.send(reply).await?;
327    Ok(())
