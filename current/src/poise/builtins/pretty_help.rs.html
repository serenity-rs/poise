3use crate::{serenity_prelude as serenity, CreateReply};

6/// Optional configuration for how the help message from [`pretty_help()`] looks
7pub struct PrettyHelpConfiguration<'a> {
8    /// Extra text displayed at the bottom of your message. Can be used for help and tips specific
9    /// to your bot
10    pub extra_text_at_bottom: &'a str,
11    /// Whether to make the response ephemeral if possible. Can be nice to reduce clutter
12    pub ephemeral: bool,
13    /// Whether to list context menu commands as well
14    pub show_context_menu_commands: bool,
15    /// Whether to list context menu commands as well
16    pub show_subcommands: bool,
17    /// Whether to include [`crate::Command::description`] (above [`crate::Command::help_text`]).
18    pub include_description: bool,
19    /// Color of the Embed
20    pub color: (u8, u8, u8),
21    #[doc(hidden)]
22    pub __non_exhaustive: (),
23}
24
25impl Default for PrettyHelpConfiguration<'_> {
26    fn default() -> Self {
27        Self {
28            extra_text_at_bottom: "",
29            ephemeral: true,
30            show_context_menu_commands: false,
31            show_subcommands: false,
32            include_description: true,
33            color: (0, 110, 51),
34            __non_exhaustive: (),
35        }
36    }
37}

39/// A help command that works similarly to `builtin::help` but outputs text in an embed.
40///
41pub async fn pretty_help<U, E>(
42    ctx: crate::Context<'_, U, E>,
43    command: Option<&str>,
44    config: PrettyHelpConfiguration<'_>,
45) -> Result<(), serenity::Error> {
46    match command {
47        Some(command) => pretty_help_single_command(ctx, command, config).await,
48        None => pretty_help_all_commands(ctx, config).await,
49    }
50}

52/// Printing an overview of all commands (e.g. `~help`)
53async fn pretty_help_all_commands<U, E>(
54    ctx: crate::Context<'_, U, E>,
55    config: PrettyHelpConfiguration<'_>,
56) -> Result<(), serenity::Error> {
57    let commands = ctx.framework().options().commands.iter().filter(|cmd| {
58        !cmd.hide_in_help
59            && (cmd.prefix_action.is_some()
60                || cmd.slash_action.is_some()
61                || (cmd.context_menu_action.is_some() && config.show_context_menu_commands))
62    });
63
64    let mut categories = indexmap::IndexMap::<Option<&str>, Vec<&crate::Command<U, E>>>::new();
65    for cmd in commands {
66        categories
67            .entry(cmd.category.as_deref())
68            .or_default()
69            .push(cmd);
70    }
71
72    let options_prefix = super::help::get_prefix_from_options(ctx).await;
73
74    let fields = categories
75        .into_iter()
76        .filter(|(_, cmds)| !cmds.is_empty())
77        .map(|(category, mut cmds)| {
78            // get context menu items at the bottom
79            cmds.sort_by_key(|cmd| cmd.slash_action.is_none() && cmd.prefix_action.is_none());
80
81            let mut buffer = String::new();
82
83            for cmd in cmds {
84                let name = cmd.context_menu_name.as_deref().unwrap_or(&cmd.name);
85                let prefix = format_cmd_prefix(cmd, &options_prefix);
86
87                if let Some(description) = cmd.description.as_deref() {
88                    writeln!(buffer, "{}{}`: *{}*", prefix, name, description).ok();
89                } else {
90                    writeln!(buffer, "{}{}`.", prefix, name).ok();
91                }
92
93                if config.show_subcommands {
94                    for sbcmd in &cmd.subcommands {
95                        let name = sbcmd.context_menu_name.as_deref().unwrap_or(&sbcmd.name);
96                        let prefix = format_cmd_prefix(sbcmd, &options_prefix);
97
98                        if let Some(description) = sbcmd.description.as_deref() {
99                            writeln!(buffer, "> {}{}`: *{}*", prefix, name, description).ok();
100                        } else {
101                            writeln!(buffer, "> {}{}`.", prefix, name).ok();
102                        }
103                    }
104                }
105            }
106            if let Some((i, _)) = buffer.char_indices().nth(1024) {
107                buffer.truncate(i);
108            }
109            (category.unwrap_or_default(), buffer, false)
110        })
111        .collect::<Vec<_>>();
112
113    let embed = serenity::CreateEmbed::new()
114        .title("Help")
115        .fields(fields)
116        .color(config.color)
117        .footer(serenity::CreateEmbedFooter::new(
118            config.extra_text_at_bottom,
119        ));
120
121    let reply = crate::CreateReply::default()
122        .embed(embed)
123        .ephemeral(config.ephemeral);
124
125    ctx.send(reply).await?;
126
127    Ok(())
128}

130/// Figures out which prefix a command should have
131fn format_cmd_prefix<U, E>(cmd: &crate::Command<U, E>, options_prefix: &Option<String>) -> String {
132    if cmd.slash_action.is_some() {
133        "`/".into()
134    } else if cmd.prefix_action.is_some() {
135        format!("`{}", options_prefix.as_deref().unwrap_or_default())
136    } else if cmd.context_menu_action.is_some() {
137        match cmd.context_menu_action {
138            Some(crate::ContextMenuCommandAction::Message(_)) => "Message menu: `".into(),
139            Some(crate::ContextMenuCommandAction::User(_)) => "User menu: `".into(),
140            Some(crate::ContextMenuCommandAction::__NonExhaustive) | None => {
141                unreachable!()
142            }
143        }
144    } else {
145        "`".into()
146    }
147}

149/// Code for printing help of a specific command (e.g. `~help my_command`)
150async fn pretty_help_single_command<U, E>(
151    ctx: crate::Context<'_, U, E>,
152    command_name: &str,
153    config: PrettyHelpConfiguration<'_>,
154) -> Result<(), serenity::Error> {
155    let commands = &ctx.framework().options().commands;
156
157    // Try interpret the command name as a context menu command first
158    let command = commands
159        .iter()
160        .find(|cmd| {
161            cmd.context_menu_name
162                .as_ref()
163                .is_some_and(|n| n.eq_ignore_ascii_case(command_name))
164        })
165        // Then interpret command name as a normal command (possibly nested subcommand)
166        .or(crate::find_command(commands, command_name, true, &mut vec![]).map(|(c, _, _)| c));
167
168    let Some(command) = command else {
169        ctx.send(
170            CreateReply::default()
171                .content(format!("No such command `{}`", command_name))
172                .ephemeral(config.ephemeral),
173        )
174        .await?;
175
176        return Ok(());
177    };
178
179    let mut invocations = Vec::new();
180    let mut subprefix = None;
181
182    if command.slash_action.is_some() {
183        invocations.push(format!("`/{}`", command.name));
184        subprefix = Some(format!("> `/{}`", command.name));
185    }
186    if command.prefix_action.is_some() {
187        let prefix = super::help::get_prefix_from_options(ctx)
188            .await
189            // This can happen if the prefix is dynamic, and the callback fails
190            // due to help being invoked with slash or context menu commands.
191            .unwrap_or_else(|| String::from("<prefix>"));
192        invocations.push(format!("`{}{}`", prefix, command.name));
193        subprefix = subprefix.or(Some(format!("> `{}{}`", prefix, command.name)));
194    }
195    if command.context_menu_name.is_some() && command.context_menu_action.is_some() {
196        let kind = match command.context_menu_action {
197            Some(crate::ContextMenuCommandAction::User(_)) => "user",
198            Some(crate::ContextMenuCommandAction::Message(_)) => "message",
199            Some(crate::ContextMenuCommandAction::__NonExhaustive) | None => unreachable!(),
200        };
201        invocations.push(format!(
202            "`{}` (on {})",
203            command
204                .context_menu_name
205                .as_deref()
206                .unwrap_or(&command.name),
207            kind
208        ));
209        subprefix = subprefix.or(Some(String::from("> ")));
210    }
211    // At least one of the three if blocks should have triggered
212    assert!(!invocations.is_empty());
213    assert!(subprefix.is_some());
214
215    let invocations = invocations
216        .into_iter()
217        .reduce(|x, y| format!("{x}\n{y}"))
218        .map(|s| ("", s, false));
219
220    let description = match (&command.description, &command.help_text) {
221        (Some(description), Some(help_text)) if config.include_description => {
222            format!("{}\n\n{}", description, help_text)
223        }
224        (_, Some(help_text)) => help_text.clone(),
225        (Some(description), None) => description.clone(),
226        (None, None) => "No help available".to_string(),
227    };
228
229    let parameters = command
230        .parameters
231        .iter()
232        .map(|parameter| {
233            let req = if parameter.required {
234                "required"
235            } else {
236                "optional"
237            };
238            if let Some(description) = parameter.description.as_deref() {
239                format!("`{}` ({}) *{} *.", parameter.name, req, description)
240            } else {
241                format!("`{}` ({}).", parameter.name, req)
242            }
243        })
244        .reduce(|x, y| format!("{x}\n{y}"))
245        .map(|s| ("Parameters", s, false));
246
247    let sbcmds = command
248        .subcommands
249        .iter()
250        .map(|sbcmd| {
251            let prefix = format_cmd_prefix(sbcmd, &subprefix); // i have no idea about this really
252            let name = sbcmd.context_menu_name.as_deref().unwrap_or(&sbcmd.name);
253            if let Some(description) = sbcmd.description.as_deref() {
254                format!("> {}{}`: *{} *", prefix, name, description)
255            } else {
256                format!("> {}{}`", prefix, name,)
257            }
258        })
259        .reduce(|x, y| format!("{x}\n{y}"))
260        .map(|s| ("Subcommands", s, false));
261
262    let fields = invocations
263        .into_iter()
264        .chain(parameters.into_iter())
265        .chain(sbcmds.into_iter());
266
267    let embed = serenity::CreateEmbed::default()
268        .description(description)
269        .fields(fields);
270
271    let reply = CreateReply::default()
272        .embed(embed)
273        .ephemeral(config.ephemeral);
274
275    ctx.send(reply).await?;
276    Ok(())
277}
