6/// Optional configuration for how the help message from [`help()`] looks
7pub struct HelpConfiguration<'a> {
8    /// Extra text displayed at the bottom of your message. Can be used for help and tips specific
9    /// to your bot
10    pub extra_text_at_bottom: &'a str,
11    /// Whether to make the response ephemeral if possible. Can be nice to reduce clutter
12    pub ephemeral: bool,
13    /// Whether to list context menu commands as well
14    pub show_context_menu_commands: bool,
15    /// Whether to list context menu commands as well
16    pub show_subcommands: bool,
17    /// Whether to include [`crate::Command::description`] (above [`crate::Command::help_text`]).
18    pub include_description: bool,
19    #[doc(hidden)]
20    pub __non_exhaustive: (),
21}
22
23impl Default for HelpConfiguration<'_> {
24    fn default() -> Self {
25        Self {
26            extra_text_at_bottom: "",
27            ephemeral: true,
28            show_context_menu_commands: false,
29            show_subcommands: false,
30            include_description: true,
31            __non_exhaustive: (),
32        }
33    }
34}

36/// Convenience function to align descriptions behind commands
37struct TwoColumnList(Vec<(String, Option<String>)>);
38
39impl TwoColumnList {
40    /// Creates a new [`TwoColumnList`]
41    fn new() -> Self {
42        Self(Vec::new())
43    }
44
45    /// Add a line that needs the padding between the columns
46    fn push_two_colums(&mut self, command: String, description: String) {
47        self.0.push((command, Some(description)));
48    }
49
50    /// Add a line that doesn't influence the first columns's width
51    fn push_heading(&mut self, category: &str) {
52        if !self.0.is_empty() {
53            self.0.push(("".to_string(), None));
54        }
55        let mut category = category.to_string();
56        category += ":";
57        self.0.push((category, None));
58    }
59
60    /// Convert the list into a string with aligned descriptions
61    fn into_string(self) -> String {
62        let longest_command = self
63            .0
64            .iter()
65            .filter_map(|(command, description)| {
66                if description.is_some() {
67                    Some(command.len())
68                } else {
69                    None
70                }
71            })
72            .max()
73            .unwrap_or(0);
74        let mut text = String::new();
75        for (command, description) in self.0 {
76            if let Some(description) = description {
77                let padding = " ".repeat(longest_command - command.len() + 3);
78                writeln!(text, "{}{}{}", command, padding, description).unwrap();
79            } else {
80                writeln!(text, "{}", command).unwrap();
81            }
82        }
83        text
84    }
85}

87/// Get the prefix from options
88pub(super) async fn get_prefix_from_options<U, E>(ctx: crate::Context<'_, U, E>) -> Option<String> {
89    let options = &ctx.framework().options().prefix_options;
90    match &options.prefix {
91        Some(fixed_prefix) => Some(fixed_prefix.clone()),
92        None => match options.dynamic_prefix {
93            Some(dynamic_prefix_callback) => {
94                match dynamic_prefix_callback(crate::PartialContext::from(ctx)).await {
95                    Ok(Some(dynamic_prefix)) => Some(dynamic_prefix),
96                    _ => None,
97                }
98            }
99            None => None,
100        },
101    }
102}

104/// Format context menu command name
105fn format_context_menu_name<U, E>(command: &crate::Command<U, E>) -> Option<String> {
106    let kind = match command.context_menu_action {
107        Some(crate::ContextMenuCommandAction::User(_)) => "user",
108        Some(crate::ContextMenuCommandAction::Message(_)) => "message",
109        Some(crate::ContextMenuCommandAction::__NonExhaustive) => unreachable!(),
110        None => return None,
111    };
112    Some(format!(
113        "{} (on {})",
114        command
115            .context_menu_name
116            .as_deref()
117            .unwrap_or(&command.name),
118        kind
119    ))
120}

122/// Code for printing help of a specific command (e.g. `~help my_command`)
123async fn help_single_command<U, E>(
124    ctx: crate::Context<'_, U, E>,
125    command_name: &str,
126    config: HelpConfiguration<'_>,
127) -> Result<(), serenity::Error> {
128    let commands = &ctx.framework().options().commands;
129    // Try interpret the command name as a context menu command first
130    let mut command = commands.iter().find(|command| {
131        if let Some(context_menu_name) = &command.context_menu_name {
132            if context_menu_name.eq_ignore_ascii_case(command_name) {
133                return true;
134            }
135        }
136        false
137    });
138    // Then interpret command name as a normal command (possibly nested subcommand)
139    if command.is_none() {
140        if let Some((c, _, _)) = crate::find_command(commands, command_name, true, &mut vec![]) {
141            command = Some(c);
142        }
143    }
144
145    let reply = if let Some(command) = command {
146        let mut invocations = Vec::new();
147        let mut subprefix = None;
148        if command.slash_action.is_some() {
149            invocations.push(format!("`/{}`", command.name));
150            subprefix = Some(format!("  /{}", command.name));
151        }
152        if command.prefix_action.is_some() {
153            let prefix = match get_prefix_from_options(ctx).await {
154                Some(prefix) => prefix,
155                // None can happen if the prefix is dynamic, and the callback
156                // fails due to help being invoked with slash or context menu
157                // commands. Not sure there's a better way to handle this.
158                None => String::from("<prefix>"),
159            };
160            invocations.push(format!("`{}{}`", prefix, command.name));
161            if subprefix.is_none() {
162                subprefix = Some(format!("  {}{}", prefix, command.name));
163            }
164        }
165        if command.context_menu_name.is_some() && command.context_menu_action.is_some() {
166            // Since command.context_menu_action is Some, this unwrap is safe
167            invocations.push(format_context_menu_name(command).unwrap());
168            if subprefix.is_none() {
169                subprefix = Some(String::from("  "));
170            }
171        }
172        // At least one of the three if blocks should have triggered
173        assert!(subprefix.is_some());
174        assert!(!invocations.is_empty());
175        let invocations = invocations.join("\n");
176
177        let mut text = match (&command.description, &command.help_text) {
178            (Some(description), Some(help_text)) => {
179                if config.include_description {
180                    format!("{}\n\n{}", description, help_text)
181                } else {
182                    help_text.clone()
183                }
184            }
185            (Some(description), None) => description.to_owned(),
186            (None, Some(help_text)) => help_text.clone(),
187            (None, None) => "No help available".to_string(),
188        };
189        if !command.parameters.is_empty() {
190            text += "\n\n```\nParameters:\n";
191            let mut parameterlist = TwoColumnList::new();
192            for parameter in &command.parameters {
193                let name = parameter.name.clone();
194                let description = parameter.description.as_deref().unwrap_or("");
195                let description = format!(
196                    "({}) {}",
197                    if parameter.required {
198                        "required"
199                    } else {
200                        "optional"
201                    },
202                    description,
203                );
204                parameterlist.push_two_colums(name, description);
205            }
206            text += &parameterlist.into_string();
207            text += "```";
208        }
209        if !command.subcommands.is_empty() {
210            text += "\n\n```\nSubcommands:\n";
211            let mut commandlist = TwoColumnList::new();
212            // Subcommands can exist on context menu commands, but there's no
213            // hierarchy in the menu, so just display them as a list without
214            // subprefix.
215            preformat_subcommands(
216                &mut commandlist,
217                command,
218                &subprefix.unwrap_or_else(|| String::from("  ")),
219            );
220            text += &commandlist.into_string();
221            text += "```";
222        }
223        format!("**{}**\n\n{}", invocations, text)
224    } else {
225        format!("No such command `{}`", command_name)
226    };
227
228    let reply = CreateReply::default()
229        .content(reply)
230        .ephemeral(config.ephemeral);
231
232    ctx.send(reply).await?;
233    Ok(())
234}

236/// Recursively formats all subcommands
237fn preformat_subcommands<U, E>(
238    commands: &mut TwoColumnList,
239    command: &crate::Command<U, E>,
240    prefix: &str,
241) {
242    let as_context_command = command.slash_action.is_none() && command.prefix_action.is_none();
243    for subcommand in &command.subcommands {
244        let command = if as_context_command {
245            let name = format_context_menu_name(subcommand);
246            if name.is_none() {
247                continue;
248            };
249            name.unwrap()
250        } else {
251            format!("{} {}", prefix, subcommand.name)
252        };
253        let description = subcommand.description.as_deref().unwrap_or("").to_string();
254        commands.push_two_colums(command, description);
255        // We could recurse here, but things can get cluttered quickly.
256        // Instead, we show (using this function) subsubcommands when
257        // the user asks for help on the subcommand.
258    }
259}

261/// Preformat lines (except for padding,) like `("  /ping", "Emits a ping message")`
262fn preformat_command<U, E>(
263    commands: &mut TwoColumnList,
264    config: &HelpConfiguration<'_>,
265    command: &crate::Command<U, E>,
266    indent: &str,
267    options_prefix: Option<&str>,
268) {
269    let prefix = if command.slash_action.is_some() {
270        String::from("/")
271    } else if command.prefix_action.is_some() {
272        options_prefix.map(String::from).unwrap_or_default()
273    } else {
274        // This is not a prefix or slash command, i.e. probably a context menu only command
275        // This should have been filtered out in `generate_all_commands`
276        unreachable!();
277    };
278
279    let prefix = format!("{}{}{}", indent, prefix, command.name);
280    commands.push_two_colums(
281        prefix.clone(),
282        command.description.as_deref().unwrap_or("").to_string(),
283    );
284    if config.show_subcommands {
285        preformat_subcommands(commands, command, &prefix)
286    }
287}

289/// Create help text for `help_all_commands`
290///
291/// This is a separate function so we can have tests for it
292async fn generate_all_commands<U, E>(
293    ctx: crate::Context<'_, U, E>,
294    config: &HelpConfiguration<'_>,
295) -> Result<String, serenity::Error> {
296    let mut categories = indexmap::IndexMap::<Option<&str>, Vec<&crate::Command<U, E>>>::new();
297    for cmd in &ctx.framework().options().commands {
298        categories
299            .entry(cmd.category.as_deref())
300            .or_default()
301            .push(cmd);
302    }
303
304    let options_prefix = get_prefix_from_options(ctx).await;
305
306    let mut menu = String::from("```\n");
307
308    let mut commandlist = TwoColumnList::new();
309    for (category_name, commands) in categories {
310        let commands = commands
311            .into_iter()
312            .filter(|cmd| {
313                !cmd.hide_in_help && (cmd.prefix_action.is_some() || cmd.slash_action.is_some())
314            })
315            .collect::<Vec<_>>();
316        if commands.is_empty() {
317            continue;
318        }
319        commandlist.push_heading(category_name.unwrap_or("Commands"));
320        for command in commands {
321            preformat_command(
322                &mut commandlist,
323                config,
324                command,
325                "  ",
326                options_prefix.as_deref(),
327            );
328        }
329    }
330    menu += &commandlist.into_string();
331
332    if config.show_context_menu_commands {
333        menu += "\nContext menu commands:\n";
334
335        for command in &ctx.framework().options().commands {
336            let name = format_context_menu_name(command);
337            if name.is_none() {
338                continue;
339            };
340            let _ = writeln!(menu, "  {}", name.unwrap());
341        }
342    }
343
344    menu += "\n";
345    menu += config.extra_text_at_bottom;
346    menu += "\n```";
347
348    Ok(menu)
349}

351/// Code for printing an overview of all commands (e.g. `~help`)
352async fn help_all_commands<U, E>(
353    ctx: crate::Context<'_, U, E>,
354    config: HelpConfiguration<'_>,
355) -> Result<(), serenity::Error> {
356    let menu = generate_all_commands(ctx, &config).await?;
357    let reply = CreateReply::default()
358        .content(menu)
359        .ephemeral(config.ephemeral);
360
361    ctx.send(reply).await?;
362    Ok(())
363}

365/// A help command that outputs text in a code block, groups commands by categories, and annotates
366/// commands with a slash if they exist as slash commands.
367///
368/// Example usage from Ferris, the Discord bot running in the Rust community server:
369/// ```rust
370/// # type Error = Box<dyn std::error::Error>;
371/// # type Context<'a> = poise::Context<'a, (), Error>;
372/// /// Show this menu
373/// #[poise::command(prefix_command, track_edits, slash_command)]
374/// pub async fn help(
375///     ctx: Context<'_>,
376///     #[description = "Specific command to show help about"] command: Option<String>,
377/// ) -> Result<(), Error> {
378///     let config = poise::builtins::HelpConfiguration {
379///         extra_text_at_bottom: "\
380/// Type ?help command for more info on a command.
381/// You can edit your message to the bot and the bot will edit its response.",
382///         ..Default::default()
383///     };
384///     poise::builtins::help(ctx, command.as_deref(), config).await?;
385///     Ok(())
386/// }
387/// ```
388/// Output:
389/// ```text
390/// Playground:
391///   ?play        Compile and run Rust code in a playground
392///   ?eval        Evaluate a single Rust expression
393///   ?miri        Run code and detect undefined behavior using Miri
394///   ?expand      Expand macros to their raw desugared form
395///   ?clippy      Catch common mistakes using the Clippy linter
396///   ?fmt         Format code using rustfmt
397///   ?microbench  Benchmark small snippets of code
398///   ?procmacro   Compile and use a procedural macro
399///   ?godbolt     View assembly using Godbolt
400///   ?mca         Run performance analysis using llvm-mca
401///   ?llvmir      View LLVM IR using Godbolt
402/// Crates:
403///   /crate       Lookup crates on crates.io
404///   /doc         Lookup documentation
405/// Moderation:
406///   /cleanup     Deletes the bot's messages for cleanup
407///   /ban         Bans another person
408///   ?move        Move a discussion to another channel
409///   /rustify     Adds the Rustacean role to members
410/// Miscellaneous:
411///   ?go          Evaluates Go code
412///   /source      Links to the bot GitHub repo
413///   /help        Show this menu
414///
415/// Type ?help command for more info on a command.
416/// You can edit your message to the bot and the bot will edit its response.
417/// ```
418pub async fn help<U, E>(
419    ctx: crate::Context<'_, U, E>,
420    command: Option<&str>,
421    config: HelpConfiguration<'_>,
422) -> Result<(), serenity::Error> {
423    match command {
424        Some(command) => help_single_command(ctx, command, config).await,
425        None => help_all_commands(ctx, config).await,
426    }
