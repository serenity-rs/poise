11/// Implement this trait on types that you want to use as a slash command parameter.
12#[async_trait::async_trait]
13pub trait SlashArgument: Sized {
14    /// Extract a Rust value of type T from the slash command argument, given via a
15    /// [`serenity::json::Value`].
16    ///
17    /// Don't call this method directly! Use [`crate::extract_slash_argument!`]
18    async fn extract(
19        ctx: &serenity::Context,
20        interaction: &serenity::CommandInteraction,
21        value: &serenity::ResolvedValue<'_>,
22    ) -> Result<Self, SlashArgError>;
23
24    /// Create a slash command parameter equivalent to type T.
25    ///
26    /// Only fields about the argument type are filled in. The caller is still responsible for
27    /// filling in `name()`, `description()`, and possibly `required()` or other fields.
28    ///
29    /// Don't call this method directly! Use [`crate::create_slash_argument!`]
30    fn create(builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption;
31
32    /// If this is a choice parameter, returns the choices
33    ///
34    /// Don't call this method directly! Use [`crate::slash_argument_choices!`]
35    fn choices() -> Vec<crate::CommandParameterChoice> {
36        Vec::new()
37    }
38}

40/// Implemented for all types that can be used as a function parameter in a slash command.
41///
42/// Currently marked `#[doc(hidden)]` because implementing this trait requires some jank due to a
43/// `PhantomData` hack and the auto-deref specialization hack.
44#[doc(hidden)]
45#[async_trait::async_trait]
46pub trait SlashArgumentHack<T>: Sized {
47    async fn extract(
48        self,
49        ctx: &serenity::Context,
50        interaction: &serenity::CommandInteraction,
51        value: &serenity::ResolvedValue<'_>,
52    ) -> Result<T, SlashArgError>;
53
54    fn create(self, builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption;
55
56    fn choices(self) -> Vec<crate::CommandParameterChoice> {
57        Vec::new()
58    }
59}

61/// Full version of [`crate::SlashArgument::extract`].
62///
63/// Uses specialization to get full coverage of types. Pass the type as the first argument
64#[macro_export]
65macro_rules! extract_slash_argument {
66    ($target:ty, $ctx:expr, $interaction:expr, $value:expr) => {{
67        use $crate::SlashArgumentHack as _;
68        (&&std::marker::PhantomData::<$target>).extract($ctx, $interaction, $value)
69    }};
70}
71/// Full version of [`crate::SlashArgument::create`].
72///
73/// Uses specialization to get full coverage of types. Pass the type as the first argument
74#[macro_export]
75macro_rules! create_slash_argument {
76    ($target:ty, $builder:expr) => {{
77        use $crate::SlashArgumentHack as _;
78        (&&std::marker::PhantomData::<$target>).create($builder)
79    }};
80}

84#[macro_export]

86    ($target:ty) => {{

92/// Handles arbitrary types that can be parsed from string.
93#[async_trait::async_trait]
94impl<T> SlashArgumentHack<T> for PhantomData<T>
95where
96    T: serenity::ArgumentConvert + Send + Sync,
97    T::Err: std::error::Error + Send + Sync + 'static,
98{
99    async fn extract(
100        self,
101        ctx: &serenity::Context,
102        interaction: &serenity::CommandInteraction,
103        value: &serenity::ResolvedValue<'_>,
104    ) -> Result<T, SlashArgError> {
105        let string = match value {
106            serenity::ResolvedValue::String(str) => *str,
107            _ => {
108                return Err(SlashArgError::CommandStructureMismatch {
109                    description: "expected string",
110                })
111            }
112        };
113
114        T::convert(
115            ctx,
116            interaction.guild_id,
117            Some(interaction.channel_id),
118            string,
119        )
120        .await
121        .map_err(|e| SlashArgError::Parse {
122            error: e.into(),
123            input: string.into(),
124        })
125    }
126
127    fn create(self, builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption {
128        builder.kind(serenity::CommandOptionType::String)
129    }
130}

132/// Implements slash argument trait for integer types
133macro_rules! impl_for_integer {
134    ($($t:ty)*) => { $(
135        #[async_trait::async_trait]
136        impl SlashArgument for $t {
137            async fn extract(
138                _: &serenity::Context,
139                _: &serenity::CommandInteraction,
140                value: &serenity::ResolvedValue<'_>,
141            ) -> Result<$t, SlashArgError> {
142                match *value {
143                    serenity::ResolvedValue::Integer(x) => x
144                        .try_into()
145                        .map_err(|_| SlashArgError::CommandStructureMismatch {
146                            description: "received out of bounds integer",
147                        }),
148                    _ => Err(SlashArgError::CommandStructureMismatch {
149                        description: "expected integer",
150                    }),
151                }
152            }
153
154            fn create(builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption {
155                builder
156                    .min_number_value(f64::max(<$t>::MIN as f64, -9007199254740991.))
157                    .max_number_value(f64::min(<$t>::MAX as f64, 9007199254740991.))
158                    .kind(serenity::CommandOptionType::Integer)
159            }
160        }
161    )* };
162}
163impl_for_integer!(i8 i16 i32 i64 isize u8 u16 u32 u64 usize);

165#[async_trait::async_trait]
166impl<T: SlashArgument + Sync> SlashArgumentHack<T> for &PhantomData<T> {
167    async fn extract(
168        self,
169        ctx: &serenity::Context,
170        interaction: &serenity::CommandInteraction,
171        value: &serenity::ResolvedValue<'_>,
172    ) -> Result<T, SlashArgError> {
173        <T as SlashArgument>::extract(ctx, interaction, value).await
174    }
175
176    fn create(self, builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption {
177        <T as SlashArgument>::create(builder)
178    }
179
180    fn choices(self) -> Vec<crate::CommandParameterChoice> {
181        <T as SlashArgument>::choices()
182    }
183}

185/// Versatile macro to implement `SlashArgumentHack` for simple types
186macro_rules! impl_slash_argument {
187    ($type:ty, |$ctx:pat, $interaction:pat, $slash_param_type:ident ( $($arg:pat),* )| $extractor:expr) => {
188        #[async_trait::async_trait]
189        impl SlashArgument for $type {
190            async fn extract(
191                $ctx: &serenity::Context,
192                $interaction: &serenity::CommandInteraction,
193                value: &serenity::ResolvedValue<'_>,
194            ) -> Result<$type, SlashArgError> {
195                match *value {
196                    serenity::ResolvedValue::$slash_param_type( $($arg),* ) => Ok( $extractor ),
197                    _ => Err(SlashArgError::CommandStructureMismatch {
198                        description: concat!("expected ", stringify!($slash_param_type))
199                    }),
200                }
201            }
202
203            fn create(builder: serenity::CreateCommandOption) -> serenity::CreateCommandOption {
204                builder.kind(serenity::CommandOptionType::$slash_param_type)
205            }
206        }
207    };
208}

236impl_slash_argument!(serenity::ChannelId, |_, _, Channel(channel)| channel.id);

239impl_slash_argument!(serenity::GuildChannel, |ctx, _, Channel(channel)| {
240    let channel_res = channel.id.to_channel(ctx).await;
241    let channel = channel_res.map_err(SlashArgError::Http)?.guild();
242    channel.ok_or(SlashArgError::Http(serenity::Error::Model(
243        serenity::ModelError::InvalidChannelType,
244    )))?
245});
