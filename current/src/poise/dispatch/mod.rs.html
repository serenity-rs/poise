13// TODO: integrate serenity::Context in here? Every place where FrameworkContext is passed is also
14// passed serenity::Context
15/// A view into data stored by [`crate::Framework`]
16pub struct FrameworkContext<'a, U, E> {
17    /// User ID of this bot
18    pub bot_id: serenity::UserId,
19    /// Framework configuration
20    pub options: &'a crate::FrameworkOptions<U, E>,
21    /// Your provided user data
22    pub user_data: &'a U,
23    /// Serenity shard manager. Can be used for example to shutdown the bot
24    pub shard_manager: &'a std::sync::Arc<serenity::ShardManager>,
25    // deliberately not non exhaustive because you need to create FrameworkContext from scratch
26    // to run your own event loop
27}
28impl<U, E> Copy for FrameworkContext<'_, U, E> {}
29impl<U, E> Clone for FrameworkContext<'_, U, E> {
30    fn clone(&self) -> Self {
31        *self
32    }
33}
34impl<'a, U, E> FrameworkContext<'a, U, E> {
35    /// Returns the stored framework options, including commands.
36    ///
37    /// This function exists for API compatiblity with [`crate::Framework`]. On this type, you can
38    /// also just access the public `options` field.
39    pub fn options(&self) -> &'a crate::FrameworkOptions<U, E> {
40        self.options
41    }
42
43    /// Returns the serenity's client shard manager.
44    ///
45    /// This function exists for API compatiblity with [`crate::Framework`]. On this type, you can
46    /// also just access the public `shard_manager` field.
47    pub fn shard_manager(&self) -> std::sync::Arc<serenity::ShardManager> {
48        self.shard_manager.clone()
49    }
50
51    /// Retrieves user data
52    ///
53    /// This function exists for API compatiblity with [`crate::Framework`]. On this type, you can
54    /// also just access the public `user_data` field.
55    #[allow(clippy::unused_async)] // for API compatibility with Framework
56    pub async fn user_data(&self) -> &'a U {
57        self.user_data
58    }
59}

61/// Central event handling function of this library
62pub async fn dispatch_event<U: Send + Sync, E>(
63    framework: crate::FrameworkContext<'_, U, E>,
64    ctx: &serenity::Context,
65    event: serenity::FullEvent,
66) {
67    match &event {
68        serenity::FullEvent::Message { new_message } => {
69            let invocation_data = tokio::sync::Mutex::new(Box::new(()) as _);
70            let mut parent_commands = Vec::new();
71            let trigger = crate::MessageDispatchTrigger::MessageCreate;
72            if let Err(error) = prefix::dispatch_message(
73                framework,
74                ctx,
75                new_message,
76                trigger,
77                &invocation_data,
78                &mut parent_commands,
79            )
80            .await
81            {
82                error.handle(framework.options).await;
83            }
84        }
85        serenity::FullEvent::MessageUpdate { event, .. } => {
86            if let Some(edit_tracker) = &framework.options.prefix_options.edit_tracker {
87                let msg = edit_tracker.write().unwrap().process_message_update(
88                    event,
89                    framework
90                        .options()
91                        .prefix_options
92                        .ignore_edits_if_not_yet_responded,
93                );
94
95                if let Some((msg, previously_tracked)) = msg {
96                    let invocation_data = tokio::sync::Mutex::new(Box::new(()) as _);
97                    let mut parent_commands = Vec::new();
98                    let trigger = match previously_tracked {
99                        true => crate::MessageDispatchTrigger::MessageEdit,
100                        false => crate::MessageDispatchTrigger::MessageEditFromInvalid,
101                    };
102                    if let Err(error) = prefix::dispatch_message(
103                        framework,
104                        ctx,
105                        &msg,
106                        trigger,
107                        &invocation_data,
108                        &mut parent_commands,
109                    )
110                    .await
111                    {
112                        error.handle(framework.options).await;
113                    }
114                }
115            }
116        }
117        serenity::FullEvent::MessageDelete {
118            deleted_message_id, ..
119        } => {
120            if let Some(edit_tracker) = &framework.options.prefix_options.edit_tracker {
121                let bot_response = edit_tracker
122                    .write()
123                    .unwrap()
124                    .process_message_delete(*deleted_message_id);
125                if let Some(bot_response) = bot_response {
126                    if let Err(e) = bot_response.delete(ctx).await {
127                        tracing::warn!("failed to delete bot response: {}", e);
128                    }
129                }
130            }
131        }
132        serenity::FullEvent::InteractionCreate {
133            interaction: serenity::Interaction::Command(interaction),
134        } => {
135            let invocation_data = tokio::sync::Mutex::new(Box::new(()) as _);
136            let mut parent_commands = Vec::new();
137            if let Err(error) = slash::dispatch_interaction(
138                framework,
139                ctx,
140                interaction,
141                &std::sync::atomic::AtomicBool::new(false),
142                &invocation_data,
143                &interaction.data.options(),
144                &mut parent_commands,
145            )
146            .await
147            {
148                error.handle(framework.options).await;
149            }
150        }
151        serenity::FullEvent::InteractionCreate {
152            interaction: serenity::Interaction::Autocomplete(interaction),
153        } => {
154            let invocation_data = tokio::sync::Mutex::new(Box::new(()) as _);
155            let mut parent_commands = Vec::new();
156            if let Err(error) = slash::dispatch_autocomplete(
157                framework,
158                ctx,
159                interaction,
160                &std::sync::atomic::AtomicBool::new(false),
161                &invocation_data,
162                &interaction.data.options(),
163                &mut parent_commands,
164            )
165            .await
166            {
167                error.handle(framework.options).await;
168            }
169        }
170        _ => {}
171    }
172
173    // Do this after the framework's Ready handling, so that get_user_data() doesnt
174    // potentially block infinitely
175    if let Err(error) =
176        (framework.options.event_handler)(ctx, &event, framework, framework.user_data).await
177    {
178        let error = crate::FrameworkError::EventHandler {
179            error,
180            ctx,
181            event: &event,
182            framework,
183        };
184        (framework.options.on_error)(error).await;
185    }
186}
