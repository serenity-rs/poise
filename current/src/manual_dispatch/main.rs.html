1//! poise::Framework handles client creation and event handling for you. Alternatively, you can
2//! do that yourself and merely forward the events you receive to poise. This example shows how.
3//!
4//! Note: this example configures no designated prefix. Mention the bot as a prefix instead. For
5//! that to work, please adjust the bot ID below to your bot, for the mention parsing to work.
6
7use poise::serenity_prelude as serenity;
8
9type Error = serenity::Error;
10
11#[poise::command(prefix_command)]
12#[allow(clippy::result_large_err)]
13async fn ping(ctx: poise::Context<'_, (), Error>) -> Result<(), Error> {
14    ctx.say("Pong!").await?;
15    Ok(())
16}
17
18struct Handler {
19    options: poise::FrameworkOptions<(), Error>,
20    shard_manager: std::sync::Mutex<Option<std::sync::Arc<serenity::ShardManager>>>,
21}
22#[serenity::async_trait]
23impl serenity::EventHandler for Handler {
24    async fn message(&self, ctx: serenity::Context, new_message: serenity::Message) {
25        // FrameworkContext contains all data that poise::Framework usually manages
26        let shard_manager = (*self.shard_manager.lock().unwrap()).clone().unwrap();
27        let framework_data = poise::FrameworkContext {
28            serenity_context: &ctx,
29            options: &self.options,
30            user_data: &(),
31            shard_manager: &shard_manager,
32        };
33
34        let event = serenity::FullEvent::Message { new_message };
35        poise::dispatch_event(framework_data, event).await;
36    }
37
38    // For slash commands or edit tracking to work, forward interaction_create and message_update
39}
40
41#[tokio::main]
42#[allow(clippy::result_large_err)]
43async fn main() -> Result<(), Error> {
44    let token = std::env::var("DISCORD_TOKEN").expect("missing DISCORD_TOKEN");
45    let intents = serenity::GatewayIntents::non_privileged();
46    let mut handler = Handler {
47        options: poise::FrameworkOptions {
48            commands: vec![ping()],
49            ..Default::default()
50        },
51        shard_manager: std::sync::Mutex::new(None),
52    };
53    poise::set_qualified_names(&mut handler.options.commands); // some setup
54
55    let handler = std::sync::Arc::new(handler);
56    let mut client = serenity::Client::builder(token, intents)
57        .event_handler_arc(handler.clone())
58        .await?;
59
60    *handler.shard_manager.lock().unwrap() = Some(client.shard_manager.clone());
61    client.start().await?;
62
63    Ok(())
64}
