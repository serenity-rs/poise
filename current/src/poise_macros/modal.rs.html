1//! Implements the #[derive(Modal)] derive macro
2
3use proc_macro::TokenStream;
4
5/// Representation of the struct attributes
6#[derive(Debug, Default, darling::FromMeta)]
7#[darling(allow_unknown_fields, default)]
8struct StructAttributes {
9    name: Option<String>,
10}
11
12/// Representation of the struct field attributes
13#[derive(Debug, Default, darling::FromMeta)]
14#[darling(allow_unknown_fields, default)]
15struct FieldAttributes {
16    name: Option<String>,
17    placeholder: Option<String>,
18    min_length: Option<u16>,
19    max_length: Option<u16>,
20    paragraph: Option<()>,
21}
22
23pub fn modal(input: syn::DeriveInput) -> Result<TokenStream, darling::Error> {
24    let fields = match input.data {
25        syn::Data::Struct(syn::DataStruct {
26            fields: syn::Fields::Named(fields),
27            ..
28        }) => fields.named,
29        _ => {
30            return Err(syn::Error::new(
31                input.ident.span(),
32                "Only structs with named fields can be used for choice parameters",
33            )
34            .into())
35        }
36    };
37
38    let struct_attrs: Vec<_> = input
39        .attrs
40        .into_iter()
41        .map(|attr| darling::ast::NestedMeta::Meta(attr.meta))
42        .collect();
43    let struct_attrs = <StructAttributes as darling::FromMeta>::from_list(&struct_attrs)?;
44
45    let mut builders = Vec::new();
46    let mut parsers = Vec::new();
47    for field in fields {
48        // Extract data from syn::Field
49        let field_attrs: Vec<_> = field
50            .attrs
51            .into_iter()
52            .map(|attr| darling::ast::NestedMeta::Meta(attr.meta))
53            .collect();
54        let field_attrs = <FieldAttributes as darling::FromMeta>::from_list(&field_attrs)?;
55        let field_ident = field.ident.unwrap();
56
57        // Create modal builder code for this field
58        let label = field_attrs.name.unwrap_or(field_ident.to_string());
59        let placeholder = field_attrs.placeholder.into_iter();
60        let required = crate::util::extract_type_parameter("Option", &field.ty).is_none();
61        let style = if field_attrs.paragraph.is_some() {
62            quote::quote!(serenity::InputTextStyle::Paragraph)
63        } else {
64            quote::quote!(serenity::InputTextStyle::Short)
65        };
66        let min_length = field_attrs.min_length.into_iter();
67        let max_length = field_attrs.max_length.into_iter();
68        builders.push(quote::quote! {
69            serenity::CreateActionRow::InputText({
70                let mut b = serenity::CreateInputText::new(#style, #label, stringify!(#field_ident));
71                if let Some(defaults) = &mut defaults {
72                    // Can use `defaults.#field_ident` directly in Edition 2021 due to more
73                    // specific closure capture rules
74                    let default = std::mem::take(&mut defaults.#field_ident);
75                    // Option::from().unwrap_or_default() dance to handle both T and Option<T>
76                    b = b.value(Option::from(default).unwrap_or_else(String::new));
77                }
78                b
79                    #( .placeholder(#placeholder) )*
80                    .required(#required)
81                    #( .min_length(#min_length) )*
82                    #( .max_length(#max_length) )*
83            }),
84        });
85
86        // Create modal parser code for this field
87        let ok_or = if required {
88            let error = format!("missing {}", field_ident);
89            Some(quote::quote! { .ok_or(#error)? })
90        } else {
91            None
92        };
93        parsers.push(quote::quote! {
94            #field_ident: poise::find_modal_text(&mut data, stringify!(#field_ident)) #ok_or,
95        });
96    }
97
98    let modal_title = struct_attrs.name.unwrap_or(input.ident.to_string());
99    let struct_ident = input.ident;
100    let (impl_generics, ty_generics, where_clause) = input.generics.split_for_impl();
101    Ok(quote::quote! { const _: () = {
102        use poise::serenity_prelude as serenity;
103        impl #impl_generics poise::Modal for #struct_ident #ty_generics #where_clause {
104            fn create(mut defaults: Option<Self>, custom_id: String) -> serenity::CreateInteractionResponse {
105                serenity::CreateInteractionResponse::Modal(
106                    serenity::CreateModal::new(custom_id, #modal_title).components(vec!{#( #builders )*})
107                )
108            }
109
110            fn parse(mut data: serenity::ModalInteractionData) -> ::std::result::Result<Self, &'static str> {
111                Ok(Self { #( #parsers )* })
112            }
113        }
114    }; }
115    .into())
116}
