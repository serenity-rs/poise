5fn quote_parameter(p: &super::CommandParameter) -> Result<proc_macro2::TokenStream, syn::Error> {
6    enum Modifier {
7        None,
8        Lazy,
9        Flag,
10        Rest,
11    }
12    let modifier = match (p.args.lazy, p.args.rest, p.args.flag) {
13        (false, false, false) => Modifier::None,
14        (true, false, false) => Modifier::Lazy,
15        (false, true, false) => Modifier::Rest,
16        (false, false, true) => Modifier::Flag,
17        _ => {
18            let message = "modifiers like #[lazy] or #[rest] currently cannot be used together";
19            return Err(syn::Error::new(p.span, message));
20        }
21    };
22    let type_ = &p.type_;
23    Ok(match modifier {
24        Modifier::Flag => {
25            if p.type_ != syn::parse_quote! { bool } {
26                return Err(syn::Error::new(p.type_.span(), "Must use bool for flags"));
27            }
28            // TODO: doesn't work for r#keywords :( I cant be arsed to fix this rn because basically
29            // nobody uses this feature anyways and I'd have to go change the macro_rules macro to
30            // not accept non-quoted idents anymore
31            let literal = proc_macro2::Literal::string(&p.name);
32            quote::quote! { #[flag] (#literal) }
33        }
34        Modifier::Lazy => quote::quote! { #[lazy] (#type_) },
35        Modifier::Rest => quote::quote! { #[rest] (#type_) },
36        Modifier::None => quote::quote! { (#type_) },
37    })
38}

40pub fn generate_prefix_action(inv: &Invocation) -> Result<proc_macro2::TokenStream, syn::Error> {
41    let param_idents = (0..inv.parameters.len())
42        .map(|i| format_ident!("poise_param_{i}"))
43        .collect::<Vec<_>>();
44    let param_specs = inv
45        .parameters
46        .iter()
47        .map(quote_parameter)
48        .collect::<Result<Vec<_>, syn::Error>>()?;
49    let wildcard_arg = match inv.args.discard_spare_arguments {
50        true => Some(quote::quote! { #[rest] (Option<String>), }),
51        false => None,
52    };
53
54    Ok(quote::quote! {
55        |ctx| Box::pin(async move {
56            let ( #( #param_idents, )* .. ) = ::poise::parse_prefix_args!(
57                ctx.serenity_context, ctx.msg, ctx.args, 0 =>
58                #( #param_specs, )*
59                #wildcard_arg
60            ).await.map_err(|(error, input)| poise::FrameworkError::new_argument_parse(
61                ctx.into(),
62                input,
63                error,
64            ))?;
65
66            let is_framework_cooldown = !ctx.command.manual_cooldowns
67                .unwrap_or_else(|| ctx.framework.options.manual_cooldowns);
68
69            if is_framework_cooldown {
70                ctx.command.cooldowns.lock().unwrap().start_cooldown(ctx.cooldown_context());
71            }
72
73            inner(ctx.into(), #( #param_idents, )* )
74                .await
75                .map_err(|error| poise::FrameworkError::new_command(
76                    ctx.into(),
77                    error,
78                ))
79        })
80    })
81}
