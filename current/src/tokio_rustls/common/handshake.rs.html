7use rustls::{ConnectionCommon, SideData};

10use crate::common::{Stream, TlsState};

12pub(crate) trait IoSession {
13    type Io;
14    type Session;
15
16    fn skip_handshake(&self) -> bool;
17    fn get_mut(&mut self) -> (&mut TlsState, &mut Self::Io, &mut Self::Session);
18    fn into_io(self) -> Self::Io;
19}
20
21pub(crate) enum MidHandshake<IS: IoSession> {
22    Handshaking(IS),
23    End,
24    Error { io: IS::Io, error: io::Error },
25}
26
27impl<IS, SD> Future for MidHandshake<IS>
28where
29    IS: IoSession + Unpin,
30    IS::Io: AsyncRead + AsyncWrite + Unpin,
31    IS::Session: DerefMut + Deref<Target = ConnectionCommon<SD>> + Unpin,
32    SD: SideData,
33{
34    type Output = Result<IS, (io::Error, IS::Io)>;
35
36    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
37        let this = self.get_mut();
38
39        let mut stream = match mem::replace(this, MidHandshake::End) {
40            MidHandshake::Handshaking(stream) => stream,
41            // Starting the handshake returned an error; fail the future immediately.
42            MidHandshake::Error { io, error } => return Poll::Ready(Err((error, io))),
43            _ => panic!("unexpected polling after handshake"),
44        };
45
46        if !stream.skip_handshake() {
47            let (state, io, session) = stream.get_mut();
48            let mut tls_stream = Stream::new(io, session).set_eof(!state.readable());
49
50            macro_rules! try_poll {
51                ( $e:expr ) => {
52                    match $e {
53                        Poll::Ready(Ok(_)) => (),
54                        Poll::Ready(Err(err)) => return Poll::Ready(Err((err, stream.into_io()))),
55                        Poll::Pending => {
56                            *this = MidHandshake::Handshaking(stream);
57                            return Poll::Pending;
58                        }
59                    }
60                };
61            }
62
63            while tls_stream.session.is_handshaking() {
64                try_poll!(tls_stream.handshake(cx));
65            }
66
67            try_poll!(Pin::new(&mut tls_stream).poll_flush(cx));
68        }
69
70        Poll::Ready(Ok(stream))
71    }
72}
